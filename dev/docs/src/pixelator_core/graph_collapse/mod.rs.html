15/// Finds low-degree nodes that look like sequencing errors of a nearby UMI,
16/// and the UMI each should be corrected to.
17///
18/// A sequencing error in a UMI creates a new node with few edges. A node is
19/// investigated when its degree is at most `max_degree`. It is treated as an
20/// error when some node at a positive even distance of at most `bfs_threshold`
21/// hops has a UMI within `hamming_threshold` symbols of its own. Even distance
22/// keeps the candidate on the investigated node's side of the bipartite graph.
23/// `max_degree` of 1 investigates only leaves.
24///
25/// The first element of the result maps each collapsed UMI to its corrected
26/// UMI. It is flat: a chain `A -> B -> H` is stored as `A -> H` and `B -> H`,
27/// and a match that would close a loop is dropped.
28///
29/// The second element counts what the search did, for this graph only:
30/// - `nodes_considered` is the number of nodes with degree at most `max_degree`.
31/// - `targets_found` and both distance distributions describe the matches that
32///   were kept, one entry per kept match. A match dropped because it would
33///   close a loop is not counted.
34/// - Each distance is that of the match as found, so `A -> B -> H` records the
35///   distances of `A -> B` and `B -> H`, not of the flattened `A -> H`.
36///
37/// `total_components` is left at 0. The graph is not known to be one
38/// component, so the caller counts components and adds them with
39/// [`CollapseStats::add_components`].
40pub fn graph_collapse<T>(
41    graph: &Graph<T>,
42    umi_mapping: &UmiToNodeIndexMapping,
43    bfs_threshold: usize,
44    hamming_threshold: u8,
45    max_degree: usize,
46) -> (HashMap<UMI, UMI>, CollapseStats)
47where
48    T: EdgeWeight + Sync,
49{
50    let mut stats = CollapseStats::default();
51
52    let mut matches: Vec<(UMI, UMI, (usize, usize))> = graph
53        .get_degrees()
54        .par_iter()
55        .enumerate()
56        .filter_map(|(node_id, &degree)| (degree <= max_degree).then_some(node_id))
57        // One queue and visited set per Rayon job, reused across its nodes,
58        // instead of a fresh pair for every search.
59        .map_init(
60            || (VecDeque::new(), HashSet::default()),
61            |scratch, node| {
62                find_collapse_target(
63                    graph,
64                    umi_mapping,
65                    node,
66                    bfs_threshold,
67                    hamming_threshold,
68                    scratch,
69                )
70                .map(|(umi, depth, hamming_dist)| {
71                    (
72                        umi_mapping.map_node_index_to_umi(node),
73                        umi,
74                        (depth, hamming_dist),
75                    )
76                })
77            },
78        )
79        .flatten()
80        .collect();
81    // Sort matches to make the output deterministic
82    matches.sort_unstable_by_key(|(original, _, _)| *original);
83
84    stats.add_considered_nodes(
85        graph
86            .get_degrees()
87            .par_iter()
88            .filter(|&&degree| degree <= max_degree)
89            .count(),
90    );
91
92    let mut targets: HashMap<UMI, UMI> = HashMap::default();
93    for (original, target, (bfs_dist, hamming_dist)) in matches {
94        // Skip the match if adding `original -> target` would create a loop.
95        if end_of_chain(&mut targets, target) != original {
96            targets.insert(original, target);
97            stats.add_target_stats(bfs_dist, hamming_dist);
98        }
99    }
100
101    let nodes: Vec<UMI> = targets.keys().copied().collect();
102    for node in nodes {
103        end_of_chain(&mut targets, node);
104    }
105    (targets, stats)
106}

108/// Follows arrows from `umi` to the node that has none, and points every node
109/// visited along the way at that end.
110fn end_of_chain(targets: &mut HashMap<UMI, UMI>, start: UMI) -> UMI {
111    let mut path = Vec::new();
112    let mut umi = start;
113    while let Some(&next) = targets.get(&umi) {
114        path.push(umi);
115        umi = next;
116    }
117    for node in path {
118        targets.insert(node, umi);
119    }
120    umi
121}

123/// Finds the nearest UMI that `node` could be a misread of, or `None` when the
124/// bounded search finds none.
125///
126/// On a match, returns `(umi, hops, hamming)`: the UMI, the number of graph
127/// hops from `node` to it, and the Hamming distance between the two UMIs.
128pub fn find_collapse_target<T>(
129    graph: &Graph<T>,
130    umi_mapping: &UmiToNodeIndexMapping,
131    node: NodeIdx,
132    bfs_threshold: usize,
133    hamming_threshold: u8,
134    scratch: &mut (VecDeque<(NodeIdx, usize)>, HashSet<NodeIdx>),
135) -> Option<(UMI, usize, usize)>
136where
137    T: EdgeWeight,
138{
139    let node_umi = umi_mapping.map_node_index_to_umi(node);
140    graph
141        .bounded_even_breadth_first_search_iter(node, Some(bfs_threshold), scratch)
142        .map(|(node_id, depth)| (umi_mapping.map_node_index_to_umi(node_id), depth))
143        .find(|&(umi, _)| within_hamming_distance(node_umi as u64, umi as u64, hamming_threshold))
144        .map(|(umi, depth)| {
145            (
146                umi,
147                depth,
148                hamming_distance_2bit(node_umi as u64, umi as u64) as usize,
149            )
150        })
151}

153pub trait GraphCollapseExt<T: EdgeWeight> {
154    /// Yields each node at a positive even distance from `start_node` as
155    /// `(node, hops)`, nearest first.
156    ///
157    /// `scratch` holds the queue and visited set, so callers can reuse their
158    /// allocations across searches. It is cleared on entry.
159    fn bounded_even_breadth_first_search_iter<'a>(
160        &'a self,
161        start_node: NodeIdx,
162        bound: Option<usize>,
163        scratch: &'a mut (VecDeque<(NodeIdx, usize)>, HashSet<NodeIdx>),
164    ) -> impl Iterator<Item = (NodeIdx, usize)> + 'a;
165}
166
167impl<T: EdgeWeight> GraphCollapseExt<T> for Graph<T> {
168    fn bounded_even_breadth_first_search_iter<'a>(
169        &'a self,
170        start_node: NodeIdx,
171        bound: Option<usize>,
172        scratch: &'a mut (VecDeque<(NodeIdx, usize)>, HashSet<NodeIdx>),
173    ) -> impl Iterator<Item = (NodeIdx, usize)> + 'a {
174        let (node_queue, visited) = scratch;
175        node_queue.clear();
176        visited.clear();
177        node_queue.push_back((start_node, 0));
178
179        iter::from_fn(move || {
180            while let Some((node, d)) = node_queue.pop_front() {
181                if visited.contains(&node) {
182                    continue;
183                }
184                visited.insert(node);
185
186                if bound.is_none_or(|bound| d < bound) {
187                    for neighbor in self.neighbors_iter(node) {
188                        if visited.contains(&neighbor) {
189                            continue;
190                        }
191                        node_queue.push_back((neighbor, d + 1));
192                    }
193                }
194
195                if d > 0 && d % 2 == 0 {
196                    return Some((node, d));
197                }
198            }
199            None
200        })
201    }
202}

303    /// Following `original -> target` must reach a node that is not itself collapsed.
304    fn collapse_pairs_have_a_cycle(targets: &HashMap<UMI, UMI>) -> bool {
305        for &start in targets.keys() {
306            let mut seen = HashSet::default();
307            let mut node = start;
308            while let Some(&target) = targets.get(&node) {
309                if !seen.insert(node) {
310                    return true;
311                }
312                node = target;
313            }
314        }
315        false
316    }
