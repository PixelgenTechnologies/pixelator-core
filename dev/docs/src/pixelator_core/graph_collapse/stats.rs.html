1//! What one graph-collapse run counted: the nodes it looked at, the matches it
2//! kept, and how far each kept match was from its node.
3//!
4//! Like the edgelist stats, values combine with [`AddAssign`], so per-component
5//! tallies fold into a run total.
6
7use std::ops::AddAssign;
8
9/// Tallies accumulated over graph-collapse searches.
10///
11/// The distance distributions are dense vectors indexed by distance, so the
12/// count of matches at distance `d` is at index `d`. A vector is only as long
13/// as the largest distance seen, and the accessors skip distances with no match.
14#[derive(Clone, Default)]
15pub struct CollapseStats {
16    /// Components searched. [`graph_collapse`](super::graph_collapse) leaves
17    /// this at 0, and the caller adds components with [`add_components`].
18    ///
19    /// [`add_components`]: Self::add_components
20    pub total_components: usize,
21    /// Kept matches. A match dropped for closing a loop is not counted.
22    pub targets_found: usize,
23    /// Nodes whose degree was at most `max_degree`, so were searched.
24    pub nodes_considered: usize,
25    /// Kept matches by graph hops from the node to its match.
26    pub bfs_distance_distribution: Vec<usize>,
27    /// Kept matches by Hamming distance between the node's UMI and its match's.
28    pub hamming_distance_distribution: Vec<usize>,
29}
30
31impl AddAssign for CollapseStats {
32    fn add_assign(&mut self, other: Self) {
33        self.total_components += other.total_components;
34        self.targets_found += other.targets_found;
35        self.nodes_considered += other.nodes_considered;
36        add_counts(
37            &mut self.bfs_distance_distribution,
38            other.bfs_distance_distribution,
39        );
40        add_counts(
41            &mut self.hamming_distance_distribution,
42            other.hamming_distance_distribution,
43        );
44    }
45}
46
47/// Adds `other` into `counts` index by index, growing `counts` if `other` is longer.
48fn add_counts(counts: &mut Vec<usize>, other: Vec<usize>) {
49    if counts.len() < other.len() {
50        counts.resize(other.len(), 0)
51    }
52    for (i, n) in other.into_iter().enumerate() {
53        counts[i] += n;
54    }
55}
56
57impl CollapseStats {
58    /// Records one kept match, `bfs_dist` hops away and `hamming_dist` symbols apart.
59    pub fn add_target_stats(&mut self, bfs_dist: usize, hamming_dist: usize) {
60        self.targets_found += 1;
61
62        if self.bfs_distance_distribution.len() < bfs_dist + 1 {
63            self.bfs_distance_distribution.resize(bfs_dist + 1, 0);
64        }
65        self.bfs_distance_distribution[bfs_dist] += 1;
66
67        if self.hamming_distance_distribution.len() < hamming_dist + 1 {
68            self.hamming_distance_distribution
69                .resize(hamming_dist + 1, 0);
70        }
71        self.hamming_distance_distribution[hamming_dist] += 1;
72    }
73
74    /// Adds `n_considered_nodes` to the count of nodes that were searched.
75    pub fn add_considered_nodes(&mut self, n_considered_nodes: usize) {
76        self.nodes_considered += n_considered_nodes;
77    }
78
79    /// Adds `n_components` to the count of components searched.
80    pub fn add_components(&mut self, n_components: usize) {
81        self.total_components += n_components;
82    }
83
84    /// `(hops, matches)` pairs in ascending hop order, for hop counts that
85    /// occurred at least once.
86    pub fn bfs_distance_counts(&self) -> Vec<(usize, usize)> {
87        self.bfs_distance_distribution
88            .iter()
89            .copied()
90            .enumerate()
91            .filter(|&(_, c)| c > 0)
92            .collect()
93    }
94
95    /// `(distance, matches)` pairs in ascending distance order, for Hamming
96    /// distances that occurred at least once.
97    pub fn hamming_distance_counts(&self) -> Vec<(usize, usize)> {
98        self.hamming_distance_distribution
99            .iter()
100            .copied()
101            .enumerate()
102            .filter(|&(_, c)| c > 0)
103            .collect()
104    }
105}

107#[cfg(test)]
108mod tests {
109    use super::*;
110
111    #[test]
112    fn default_is_all_zero() {
113        let stats = CollapseStats::default();
114
115        assert_eq!(stats.total_components, 0);
116        assert_eq!(stats.targets_found, 0);
117        assert_eq!(stats.nodes_considered, 0);
118        assert!(stats.bfs_distance_counts().is_empty());
119        assert!(stats.hamming_distance_counts().is_empty());
120    }
121
122    #[test]
123    fn add_target_stats_tallies_each_distance_separately() {
124        let mut stats = CollapseStats::default();
125        stats.add_target_stats(2, 1);
126        stats.add_target_stats(2, 2);
127        stats.add_target_stats(4, 1);
128
129        assert_eq!(stats.targets_found, 3);
130        assert_eq!(stats.bfs_distance_counts(), [(2, 2), (4, 1)]);
131        assert_eq!(stats.hamming_distance_counts(), [(1, 2), (2, 1)]);
132    }
133
134    /// The vectors are indexed by distance, so the unused slots below the
135    /// first match must not show up as matches.
136    #[test]
137    fn counts_skip_distances_that_never_occurred() {
138        let mut stats = CollapseStats::default();
139        stats.add_target_stats(6, 3);
140
141        assert_eq!(stats.bfs_distance_counts(), [(6, 1)]);
142        assert_eq!(stats.hamming_distance_counts(), [(3, 1)]);
143    }
144
145    #[test]
146    fn counts_come_out_in_ascending_order_whatever_order_they_arrived_in() {
147        let mut stats = CollapseStats::default();
148        stats.add_target_stats(6, 3);
149        stats.add_target_stats(2, 1);
150
151        assert_eq!(stats.bfs_distance_counts(), [(2, 1), (6, 1)]);
152        assert_eq!(stats.hamming_distance_counts(), [(1, 1), (3, 1)]);
153    }
154
155    #[test]
156    fn node_and_component_counts_accumulate() {
157        let mut stats = CollapseStats::default();
158        stats.add_considered_nodes(3);
159        stats.add_considered_nodes(4);
160        stats.add_components(1);
161        stats.add_components(2);
162
163        assert_eq!(stats.nodes_considered, 7);
164        assert_eq!(stats.total_components, 3);
165        assert_eq!(stats.targets_found, 0);
166    }
167
168    #[test]
169    fn add_assign_merges_each_field_independently() {
170        let mut total = CollapseStats::default();
171        total.add_components(1);
172        total.add_considered_nodes(5);
173        total.add_target_stats(2, 1);
174
175        // Longer on both distributions than `total`, so merging has to grow them.
176        let mut other = CollapseStats::default();
177        other.add_components(2);
178        other.add_considered_nodes(2);
179        other.add_target_stats(2, 2);
180        other.add_target_stats(4, 1);
181        total += other;
182
183        assert_eq!(total.total_components, 3);
184        assert_eq!(total.nodes_considered, 7);
185        assert_eq!(total.targets_found, 3);
186        assert_eq!(total.bfs_distance_counts(), [(2, 2), (4, 1)]);
187        assert_eq!(total.hamming_distance_counts(), [(1, 2), (2, 1)]);
188    }
189
190    /// The shorter side must not truncate the longer one.
191    #[test]
192    fn add_assign_keeps_a_longer_left_side() {
193        let mut total = CollapseStats::default();
194        total.add_target_stats(4, 3);
195
196        let mut other = CollapseStats::default();
197        other.add_target_stats(2, 1);
198        total += other;
199
200        assert_eq!(total.bfs_distance_counts(), [(2, 1), (4, 1)]);
201        assert_eq!(total.hamming_distance_counts(), [(1, 1), (3, 1)]);
202    }
203
204    #[test]
205    fn adding_an_empty_default_changes_nothing() {
206        let mut total = CollapseStats::default();
207        total.add_components(1);
208        total.add_considered_nodes(4);
209        total.add_target_stats(2, 1);
210        total += CollapseStats::default();
211
212        assert_eq!(total.total_components, 1);
213        assert_eq!(total.nodes_considered, 4);
214        assert_eq!(total.targets_found, 1);
215        assert_eq!(total.bfs_distance_counts(), [(2, 1)]);
216        assert_eq!(total.hamming_distance_counts(), [(1, 1)]);
217    }
