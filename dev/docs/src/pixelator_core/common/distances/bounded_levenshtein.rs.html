1//! Single-word Myers bit-parallel bounded infix Levenshtein search.
2//!
3//! The pattern and text are both 2-bit-packed DNA (≤ 32 bases each, one `u64`
4//! word; same little-endian packing as [`crate::encoding::encode_dna_2bit`]).
5//! The search reports whether the *whole* pattern aligns to *any substring* of
6//! the text within a bounded number of edits (substitutions + indels), i.e. an
7//! infix (text-local) Levenshtein match with free end-gaps on the text side only.
8//!
9//! # Algorithm
10//!
11//! This is Myers' 1999 infix (semi-global) approximate string matching: after
12//! column `j`, `score` is the edit distance from the whole pattern to the best
13//! text substring *ending* at `j`. We short-circuit as soon as
14//! `score <= max_edits`. The horizontal shift is a plain `<< 1` with no `| 1`,
15//! so row 0 stays 0 (report-all-occurrences / infix form).
16//!
17//! The bit-vector recurrences follow Hyyrö's D0 reformulation of Myers (same
18//! algorithm, different spelling). This is **not** Wu–Manber/`agrep` NFA
19//! matching, **not** global end-to-end Levenshtein, and **not** Damerau
20//! (no transpositions).
21//!
22//! # References
23//!
24//! - Myers, G. (1999). A fast bit-vector algorithm for approximate string
25//!   matching based on dynamic programming. *J. ACM* 46(3):395–415.
26//!   <https://doi.org/10.1145/316542.316550>
27//! - Hyyrö, H. (2001). Explaining and Extending the Bit-Parallel Approximate
28//!   String Matching Algorithm of Myers. University of Tampere, Dept. of
29//!   Computer and Information Sciences, Report A-2001-10.
30//!   <https://researchportal.tuni.fi/en/publications/explaining-and-extending-the-bit-parallel-approximate-string-matc/>
31//!
32//! # Name map (ours → Myers 1999 → Hyyrö 2001)
33//!
34//! - `peq` → Peq → Peq
35//! - `vp` → Pv → VP
36//! - `vn` → Mv → VN
37//! - `eq` → Eq → (pattern-match bits for the current text base)
38//! - `x` → Xv → Xv
39//! - `d0` → (related to Xh; D0 ≈ Xv | Xh) → D0
40//! - `hp` → Ph → HP
41//! - `hn` → Mh → HN
42//! - `score` → Score → D[m, j]
43
44/// Returns whether the packed `pattern` (low `2 * pattern_len` bits, base 0 in
45/// the lowest bits) aligns to any substring of the packed `text` (low
46/// `2 * text_len` bits) within `max_edits` edits (infix Levenshtein;
47/// substitutions + indels).
48///
49/// Both `pattern_len` and `text_len` must be ≤ 32 (one `u64` of 2-bit codes).
50///
51/// Instead of filling an `(m+1)×(n+1)` DP table, each column's vertical delta
52/// bits are packed into a `u64` and advanced one text base at a time with
53/// word-wide bit ops — O(n) for patterns of length ≤ 32. See the module docs
54/// for the Myers / Hyyrö citations and name map.
55pub fn within_bounded_levenshtein(
56    pattern: u64,
57    pattern_len: usize,
58    text: u64,
59    text_len: usize,
60    max_edits: u32,
61) -> bool {
62    debug_assert!(pattern_len <= 32 && text_len <= 32);
63
64    let m = pattern_len;
65    // Empty pattern aligns to the empty text substring with zero edits.
66    if m == 0 {
67        return true;
68    }
69
70    // `peq[c]` has bit `i` set iff pattern base `i` has 2-bit code `c`.
71    let mut peq = [0u64; 4];
72    for i in 0..m {
73        let c = ((pattern >> (2 * i)) & 0b11) as usize;
74        peq[c] |= 1u64 << i;
75    }
76
77    // All vectors are kept to `m` bits; masking after every `!`/`<<` keeps the
78    // high garbage out of the `(Eq & VP) + VP` carry chain, which otherwise
79    // wraps around the word and corrupts the low bits.
80    let mask: u64 = if m >= 64 { u64::MAX } else { (1u64 << m) - 1 };
81    let top_bit = 1u64 << (m - 1);
82    let mut vp: u64 = mask;
83    let mut vn: u64 = 0;
84    // Score against the empty text prefix: every pattern base is a deletion.
85    // I.e. the score is the number of deletions needed to align the pattern to the empty text.
86    let mut score = m as u32;
87    if score <= max_edits {
88        return true;
89    }
90
91    let n = text_len;
92    for j in 0..n {
93        // Canonical Myers "report all occurrences" (infix) step: the row-0
94        // boundary stays 0, so the horizontal shift is a plain `<< 1` with no
95        // `| 1` (that extra bit is only for whole-string edit distance).
96        // Hyyrö D0 form: x = eq | vn; d0 = ((vp + (x & vp)) ^ vp) | x.
97        let code = ((text >> (2 * j)) & 0b11) as usize;
98        let eq = peq[code];
99        let x = eq | vn;
100        let d0 = ((vp.wrapping_add(x & vp)) ^ vp) | x;
101        let hn = vp & d0;
102        let hp = vn | !(d0 | vp);
103
104        if hp & top_bit != 0 {
105            score += 1;
106        } else if hn & top_bit != 0 {
107            score -= 1;
108        }
109
110        if score <= max_edits {
111            return true;
112        }
113
114        // The bottom-row score moves by at most one per column, so if even
115        // driving it down every remaining column cannot reach the budget,
116        // there is nothing left to find.
117        let remaining = (n - j - 1) as u32;
118        if score - max_edits > remaining {
119            return false;
120        }
121
122        let hp_shifted = (hp << 1) & mask;
123        vn = hp_shifted & d0;
124        vp = ((hn << 1) | !(hp_shifted | d0)) & mask;
125    }
126
127    false
128}

130#[cfg(test)]
131mod tests {
132    use super::*;
133    use crate::encoding::encode_dna_2bit;
134
135    fn matches(pattern: &[u8], text: &[u8], k: u32) -> bool {
136        within_bounded_levenshtein(
137            encode_dna_2bit(pattern),
138            pattern.len(),
139            encode_dna_2bit(text),
140            text.len(),
141            k,
142        )
143    }
144
145    #[test]
146    fn exact_substring_matches_at_zero_edits() {
147        assert!(matches(b"ACGTACGT", b"ACGTACGT", 0));
148        assert!(matches(b"ACGTACGT", b"TTTTACGTACGTTTT", 0));
149    }
150
151    #[test]
152    fn single_substitution_needs_one_edit() {
153        // Best substring "ACGTACCT" is one substitution from the pattern.
154        assert!(!matches(b"ACGTACGT", b"TTTTACGTACCTTTT", 0));
155        assert!(matches(b"ACGTACGT", b"TTTTACGTACCTTTT", 1));
156    }
157
158    #[test]
159    fn single_deletion_needs_one_edit() {
160        // Text "ACT" is the pattern "ACGT" with the G deleted.
161        assert!(!matches(b"ACGT", b"ACT", 0));
162        assert!(matches(b"ACGT", b"ACT", 1));
163    }
164
165    #[test]
166    fn single_insertion_needs_one_edit() {
167        // Text substring "ACAGT" is the pattern "ACGT" with an extra A inserted;
168        // "ACGT" itself is not a contiguous substring here.
169        assert!(!matches(b"ACGT", b"GGACAGTGG", 0));
170        assert!(matches(b"ACGT", b"GGACAGTGG", 1));
171    }
172
173    #[test]
174    fn distant_pattern_rejected_under_small_budget() {
175        // "ACGTACGT" vs an all-G text: G matches only at pattern indices 2 and 6,
176        // so at least six substitutions are required.
177        assert!(!matches(b"ACGTACGT", b"GGGGGGGG", 1));
178        assert!(!matches(b"ACGTACGT", b"GGGGGGGG", 5));
179        assert!(matches(b"ACGTACGT", b"GGGGGGGG", 6));
180    }
181
182    #[test]
183    fn empty_text_needs_full_length_deletions() {
184        // With no text, the only alignment deletes every pattern base.
185        assert!(!matches(b"ACGT", b"", 3));
186        assert!(matches(b"ACGT", b"", 4));
187    }
188
189    #[test]
190    fn empty_pattern_always_matches() {
191        // Empty pattern aligns to the empty substring with zero edits.
192        assert!(matches(b"", b"", 0));
193        assert!(matches(b"", b"ACGT", 0));
194    }
195
196    #[test]
197    fn budget_at_least_pattern_length_matches_without_scan() {
198        // Score starts at m; if max_edits >= m the empty alignment already wins.
199        // Homopolymer vs its complement has infix distance m, so k = m-1 fails.
200        assert!(!matches(b"AAAA", b"CCCC", 3));
201        assert!(matches(b"AAAA", b"CCCC", 4));
202    }
203
204    #[test]
205    fn exact_match_at_text_start_and_end() {
206        assert!(matches(b"ACGT", b"ACGTTTTT", 0));
207        assert!(matches(b"ACGT", b"TTTTACGT", 0));
208        // Near-miss only at the opposite end still fails at zero edits.
209        assert!(!matches(b"ACGT", b"ACGCTTTT", 0));
210        assert!(!matches(b"ACGT", b"TTTTACGC", 0));
211    }
212
213    #[test]
214    fn two_substitutions_need_two_edits() {
215        // Best substring "ACCTACCT" differs from "ACGTACGT" at two sites.
216        assert!(!matches(b"ACGTACGT", b"TTTTACCTACCTTTTT", 1));
217        assert!(matches(b"ACGTACGT", b"TTTTACCTACCTTTTT", 2));
218    }
219
220    #[test]
221    fn substitution_plus_deletion_needs_two_edits() {
222        // "ATT" from "ACGT": delete C and substitute G→T (length gap blocks any
223        // single-edit path).
224        assert!(!matches(b"ACGT", b"GGATTGG", 1));
225        assert!(matches(b"ACGT", b"GGATTGG", 2));
226    }
227
228    #[test]
229    fn length_32_exact_and_near_miss() {
230        let pattern = b"ACGTACGTACGTACGTACGTACGTACGTACGT";
231        assert_eq!(pattern.len(), 32);
232        assert!(matches(pattern, pattern, 0));
233
234        // One substitution in the middle of an otherwise exact text.
235        let mut near = pattern.to_vec();
236        near[16] = b'T'; // was A
237        assert!(!matches(pattern, &near, 0));
238        assert!(matches(pattern, &near, 1));
239
240        // Unrelated length-32 text needs many edits; reject under a tight budget.
241        let distant = b"GGGGGGGGGGGGGGGGGGGGGGGGGGGGGGGG";
242        assert!(!matches(pattern, distant, 1));
243    }
244
245    #[test]
246    fn repeated_bases_exercise_multi_bit_peq() {
247        // Homopolymer pattern: peq[A] has several bits set.
248        assert!(matches(b"AAAA", b"AAAA", 0));
249        // One deletion short of the pattern.
250        assert!(!matches(b"AAAA", b"AAA", 0));
251        assert!(matches(b"AAAA", b"AAA", 1));
252        // One insertion; exact "AAAA" is still a substring of "AAAAA".
253        assert!(matches(b"AAAA", b"AAAAA", 0));
254    }
255
256    #[test]
257    fn infix_match_ignores_flanking_text() {
258        // Pattern is an interior exact substring; free end-gaps on the text side.
259        assert!(matches(b"ACGT", b"TTTTACGTAAAA", 0));
260        // Same pattern is not a prefix or suffix of the full text.
261        assert!(!b"TTTTACGTAAAA".starts_with(b"ACGT"));
262        assert!(!b"TTTTACGTAAAA".ends_with(b"ACGT"));
263    }
