10/// An antibody panel indexed for fast marker lookups by barcode sequence.
11///
12/// Each marker in a panel is associated with two barcode sequences. The panel
13/// keeps a separate map for each, mapping the 2-bit-encoded sequence (see
14/// [`encode_dna_2bit`]) to the marker identifier so that a barcode observed in
15/// the data can be resolved back to its marker in O(1).
16///
17/// Both sequences are stored exactly as they appear in the panel file. Callers
18/// are responsible for encoding an observed barcode into the same orientation
19/// before looking it up (e.g. reverse-complementing a `sequence_2` barcode read
20/// off the reverse read via [`encode_dna_2bit_rc`](crate::encoding::encode_dna_2bit_rc)).
21///
22/// Because the maps are keyed by encoded sequence, each sequence must be unique
23/// within its column; [`Panel::from_path`] rejects panels where two markers
24/// share a `sequence_1` or a `sequence_2`.
25#[derive(Clone)]
26pub struct Panel {
27    /// All markers, in the order they were loaded from the panel file. This is
28    /// the source of truth for enumerating markers/sequences; the lookup maps
29    /// below are lossy (encoded, deduped) and cannot reconstruct it.
30    markers: Vec<PanelMarker>,
31    /// Encoded `sequence_1` -> index of the marker in `markers`.
32    map_seq_1: HashMap<u64, i32>,
33    /// Encoded `sequence_2` -> index of the marker in `markers`.
34    map_seq_2: HashMap<u64, i32>,
35}
36
37/// A single marker loaded from a panel: its identifier and its two barcode
38/// sequences, stored exactly as they appear in the panel file.
39#[derive(Debug, Clone, PartialEq, Eq)]
40pub struct PanelMarker {
41    /// Human-readable marker identifier (e.g. `CD11b`).
42    pub marker_id: String,
43    /// First barcode sequence for the marker.
44    pub sequence_1: String,
45    /// Second barcode sequence for the marker.
46    pub sequence_2: String,
47}

49/// A single deserialized row of a panel CSV.
50///
51/// Only the fields consumed by [`Panel`] are declared here; any other columns
52/// present in the file (for example `control`, `full_name`, `alt_id`,
53/// `uniprot_id`) are matched by header name and ignored.
54#[derive(Debug, Deserialize)]
55struct Record {
56    /// Human-readable marker identifier (e.g. `CD11b`).
57    marker_id: String,
58    /// First barcode sequence for the marker.
59    sequence_1: String,
60    /// Second barcode sequence for the marker.
61    sequence_2: String,
62}
63
64impl Panel {
65    /// Loads a panel from a CSV file at `path`.
66    ///
67    /// The file is expected to have a header row containing at least the
68    /// `marker_id`, `sequence_1`, and `sequence_2` columns. Lines beginning with
69    /// `#` are treated as comments and skipped, which allows the leading YAML
70    /// metadata block present in the standard panel files.
71    ///
72    /// Each row's `sequence_1` and `sequence_2` are 2-bit encoded via
73    /// [`encode_dna_2bit`] exactly as they appear in the file, and stored as
74    /// keys pointing to the row's index in `markers`.
75    ///
76    /// Real panel files are ragged: rows commonly omit trailing empty columns
77    /// (for example `sample_hashing`). Short rows are padded to the header width
78    /// so header-based deserialization does not fail on the missing columns.
79    ///
80    /// # Errors
81    ///
82    /// Returns an error if the file cannot be opened or read, as soon as any
83    /// row fails to deserialize, or if a `sequence_1` or `sequence_2` is
84    /// duplicated across markers.
85    pub fn from_path<P: AsRef<Path>>(path: P) -> Result<Self, Box<dyn Error>> {
86        let mut reader = ReaderBuilder::new()
87            .comment(Some(b'#'))
88            .flexible(true)
89            .from_path(path)?;
90
91        let headers = reader.headers()?.clone();
92
93        let mut markers = Vec::new();
94
95        for result in reader.records() {
96            let mut row = result?;
97            while row.len() < headers.len() {
98                row.push_field("");
99            }
100            let record: Record = row.deserialize(Some(&headers))?;
101            markers.push(PanelMarker {
102                marker_id: record.marker_id,
103                sequence_1: record.sequence_1,
104                sequence_2: record.sequence_2,
105            });
106        }
107
108        Self::from_markers(markers)
109    }
110
111    /// Builds a panel from markers already in memory, indexing markers as they appear in
112    /// the input vector.
113    pub fn from_markers(markers: Vec<PanelMarker>) -> Result<Self, Box<dyn Error>> {
114        let mut map_seq_1 = HashMap::default();
115        let mut map_seq_2 = HashMap::default();
116        for (index, m) in markers.iter().enumerate() {
117            let index = index as i32;
118            if map_seq_1
119                .insert(encode_dna_2bit(m.sequence_1.as_bytes()), index)
120                .is_some()
121            {
122                return Err(format!(
123                    "Duplicate sequence_1 '{}' found in panel (marker {})",
124                    m.sequence_1, m.marker_id
125                )
126                .into());
127            }
128            if map_seq_2
129                .insert(encode_dna_2bit(m.sequence_2.as_bytes()), index)
130                .is_some()
131            {
132                return Err(format!(
133                    "Duplicate sequence_2 '{}' found in panel (marker {})",
134                    m.sequence_2, m.marker_id
135                )
136                .into());
137            }
138        }
139
140        Ok(Self {
141            markers,
142            map_seq_1,
143            map_seq_2,
144        })
145    }
146
147    /// Returns the index (position in [`markers`](Self::markers)) of the marker
148    /// whose `sequence_1` matches the 2-bit-encoded `seq`, or `None` if no marker
149    /// in the panel uses that sequence.
150    pub fn marker_index_seq1(&self, seq: u64) -> Option<i32> {
151        self.map_seq_1.get(&seq).copied()
152    }
153
154    /// Returns the index (position in [`markers`](Self::markers)) of the marker
155    /// whose `sequence_2` matches the 2-bit-encoded `seq`, or `None` if no marker
156    /// in the panel uses that sequence.
157    pub fn marker_index_seq2(&self, seq: u64) -> Option<i32> {
158        self.map_seq_2.get(&seq).copied()
159    }
160
161    /// Returns the `marker_id` for a 2-bit-encoded `sequence_1` barcode, or
162    /// `None` if no marker in the panel uses that sequence.
163    pub fn lookup_seq1(&self, seq: u64) -> Option<&str> {
164        self.marker_index_seq1(seq)
165            .map(|idx| self.markers[idx as usize].marker_id.as_str())
166    }
167
168    /// Returns the `marker_id` for a 2-bit-encoded `sequence_2` barcode, or
169    /// `None` if no marker in the panel uses that sequence.
170    pub fn lookup_seq2(&self, seq: u64) -> Option<&str> {
171        self.marker_index_seq2(seq)
172            .map(|idx| self.markers[idx as usize].marker_id.as_str())
173    }
174
175    /// Returns all markers loaded from the panel, in file order.
176    pub fn markers(&self) -> &[PanelMarker] {
177        &self.markers
178    }
179}

181/// Largest `mismatch_threshold` at which the fuzzy fallback is guaranteed never
182/// to *silently* assign a barcode to the wrong marker.
183///
184/// # Background
185///
186/// A barcode read off the sequencer can carry substitution errors, so an
187/// observed sequence often does not match any panel barcode exactly. We model
188/// the number of errors as the Hamming distance (count of differing bases)
189/// between the observed barcode and a panel barcode, and try to map a noisy read
190/// back to the marker it most likely came from.
191///
192/// This is only possible because panels are *designed* with their barcodes
193/// spread apart: any two differ by at least `d_min` bases (here `d_min = 3`).
194/// That spacing is what makes the barcode self-correcting for up to 1 error.
195///
196/// # How the spacing bounds the safe threshold
197///
198/// Say a read is `t` mismatches from its true marker `A`. By the triangle
199/// inequality for Hamming distance, its distance to any *other* marker `B` is
200///
201/// ```text
202/// d(read, B) ≥ d(A, B) − d(A, read) ≥ d_min − t
203/// ```
204///
205/// so the nearest a wrong marker can ever be is `d_min − t`. Weighing that
206/// against how close the read is to the right marker (`≤ t`) gives three
207/// regimes:
208///
209/// - `d_min > 2t`: the true marker is *strictly* closest — correction is
210///   unambiguous.
211/// - `d_min = 2t`: a wrong marker can *tie* the true one but never beat it — so
212///   detecting and dropping ties is enough to stay correct.
213/// - `d_min < 2t`: a wrong marker can be *strictly* closest — the read is
214///   silently misassigned, and no tie check can catch it.
215///
216/// For `d_min = 3` this makes `t = 1` the largest safe threshold: at `t = 2` a
217/// read could be at d_min - t = 1 distance away from an other marker, namely a
218/// confident wrong answer. That is the boundary [`PanelSearcher::new`] warns past.
219///
220/// # Why only direct distances matter, not reverse complements
221///
222/// The bound above concerns distances between barcodes *as stored in the panel*;
223/// reverse complements never enter it. A reverse-strand read is
224/// reverse-complemented into the panel's orientation (via
225/// [`encode_dna_2bit_rc`](crate::encoding::encode_dna_2bit_rc)) *before* lookup,
226/// based on the strand it came from, so no reverse-complemented sequence is ever
227/// added to the set of match targets. A panel barcode that happens to lie near
228/// the reverse complement of another is therefore harmless — that form is never
229/// searched. (In `proxiome-v2-immuno-155-v2.0` one barcode is only 1 mismatch
230/// from another's reverse complement, yet the minimum *direct* distance is
231/// still 4.)
232const SAFE_MISMATCH_THRESHOLD: u8 = 1;
233
234/// Outcome of resolving an observed barcode against the panel.
235#[derive(Clone, Copy, Debug, PartialEq, Eq)]
236pub enum MarkerMatch {
237    /// Resolved to a single marker (index into the panel's markers).
238    Hit(i32),
239    /// No panel marker within the mismatch threshold.
240    Miss,
241    /// Two or more markers tied for the nearest distance; rejected as unsafe.
242    Ambiguous,
243}
244
245/// Resolves an observed barcode to a panel marker, tolerating up to
246/// `mismatch_threshold` mismatches.
247///
248/// A lookup first tries an exact hash-map hit (the common case). On a miss, the
249/// fuzzy fallback is picked to fit the safety regimes documented on
250/// `SAFE_MISMATCH_THRESHOLD`:
251///
252/// - **0**: no fuzzy fallback; exact matches only (`Miss` on miss).
253/// - **1**: return the *first* marker within range (`search_first_hit`). With
254///   `d_min ≥ 3`, at most one marker can be within 1 mismatch, so the first hit
255///   is necessarily the only hit and there is nothing to disambiguate.
256/// - **≥ 2**: two markers can now be within range at once, so scan them all,
257///   keep the single closest, and return [`MarkerMatch::Ambiguous`] if the two
258///   nearest tie (`search_best_match`).
259///
260/// [`PanelSearcher::new`] warns when `mismatch_threshold` exceeds
261/// `SAFE_MISMATCH_THRESHOLD`: past that point the tie check is no longer enough,
262/// because a wrong marker can be strictly closest and slip through as a
263/// confident but incorrect match.
264#[derive(Clone)]
265pub struct PanelSearcher {
266    panel: Panel,
267    mismatch_threshold: u8,
268    /// Contiguous `(encoded sequence, marker index)` pairs for the fuzzy
269    /// fallback scan. Iterating a packed `Vec` (cache-resident and
270    /// autovectorizable) is far faster than iterating the `HashMap`, which is
271    /// kept only for the O(1) exact-hit fast path.
272    scan_seq_1: Vec<(u64, i32)>,
273    scan_seq_2: Vec<(u64, i32)>,
274}
275
276impl PanelSearcher {
277    /// Builds a searcher over `panel` that tolerates up to `mismatch_threshold`
278    /// mismatches when resolving a barcode.
279    ///
280    /// A `mismatch_threshold` above `SAFE_MISMATCH_THRESHOLD` exceeds the
281    /// collision-free guarantee the panel is designed around (see the type-level
282    /// docs), so a warning is logged: fuzzy matches may assign a read to a
283    /// different marker than intended.
284    pub fn new(panel: Panel, mismatch_threshold: u8) -> Self {
285        if mismatch_threshold > SAFE_MISMATCH_THRESHOLD {
286            log::warn!(
287                "PanelSearcher mismatch_threshold {mismatch_threshold} exceeds the safe limit of \
288                 {SAFE_MISMATCH_THRESHOLD}; within that radius, all barcodes are guaranteed to be \
289                 self-correcting, but beyond it a query can be strictly closest to the wrong marker \
290                 and be silently misassigned."
291            );
292        }
293        let scan_seq_1 = panel.map_seq_1.iter().map(|(&s, &i)| (s, i)).collect();
294        let scan_seq_2 = panel.map_seq_2.iter().map(|(&s, &i)| (s, i)).collect();
295        Self {
296            panel,
297            mismatch_threshold,
298            scan_seq_1,
299            scan_seq_2,
300        }
301    }
302
303    /// Returns the underlying panel.
304    pub fn panel(&self) -> &Panel {
305        &self.panel
306    }
307
308    pub fn search_seq1(&self, seq: u64) -> MarkerMatch {
309        self.search(seq, &self.panel.map_seq_1, &self.scan_seq_1)
310    }
311
312    pub fn search_seq2(&self, seq: u64) -> MarkerMatch {
313        self.search(seq, &self.panel.map_seq_2, &self.scan_seq_2)
314    }
315
316    /// Exact hash-map lookup, then a threshold-dependent fuzzy fallback:
317    /// first-hit at threshold 1, best-match with tie rejection at threshold ≥ 2.
318    fn search(&self, seq: u64, panel_map: &HashMap<u64, i32>, scan: &[(u64, i32)]) -> MarkerMatch {
319        if let Some(index) = panel_map.get(&seq) {
320            return MarkerMatch::Hit(*index);
321        }
322
323        // Dispatch by threshold. See `SAFE_MISMATCH_THRESHOLD`: at threshold 1 at
324        // most one marker can be in range, so the first hit is the only hit; at
325        // ≥ 2 several can be in range, so we must find the unique nearest and
326        // reject ties.
327        match self.mismatch_threshold {
328            0 => MarkerMatch::Miss,
329            1 => self.search_first_hit(seq, scan),
330            _ => self.search_best_match(seq, scan),
331        }
332    }
333
334    fn search_first_hit(&self, seq: u64, scan: &[(u64, i32)]) -> MarkerMatch {
335        scan.iter()
336            .find(|&&(marker_seq, _)| {
337                within_hamming_distance(seq, marker_seq, self.mismatch_threshold)
338            })
339            .map(|&(_, index)| MarkerMatch::Hit(index))
340            .unwrap_or(MarkerMatch::Miss)
341    }
342
343    fn search_best_match(&self, seq: u64, scan: &[(u64, i32)]) -> MarkerMatch {
344        let mut best_index = None;
345        let mut best_distance = self.mismatch_threshold + 1;
346        let mut tie = false;
347        for &(marker_seq, index) in scan {
348            let distance = hamming_distance_2bit(seq, marker_seq);
349            if distance > self.mismatch_threshold {
350                continue;
351            }
352            if distance < best_distance {
353                best_distance = distance;
354                best_index = Some(index);
355                tie = false;
356            } else if distance == best_distance {
357                tie = true;
358            }
359        }
360        match best_index {
361            Some(index) if !tie => MarkerMatch::Hit(index),
362            Some(_) => MarkerMatch::Ambiguous,
363            None => MarkerMatch::Miss,
364        }
365    }
366}
