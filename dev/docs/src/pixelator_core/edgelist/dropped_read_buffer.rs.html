13/// Column buffers for the reads the extractor discarded, flushed into the
14/// optional dropped-reads Parquet that sits next to the edgelist.
15///
16/// One row per input molecule, exactly as it appeared in the FASTQ: a single-end
17/// read fills `sequence`/`quality` and leaves the mate columns null; a pair puts
18/// R1 in `sequence`/`quality` and R2 in `sequence_2`/`quality_2`. `reason` is
19/// the [`DropReason`] that fired first.
20#[derive(Clone)]
21pub struct DroppedReadBuffer {
22    read_name: StringColumn,
23    sequence: StringColumn,
24    quality: StringColumn,
25    sequence_2: StringColumn,
26    quality_2: StringColumn,
27    /// Keys into `reason_dict`, i.e. [`DropReason::index`].
28    reason: Vec<i32>,
29    /// The names of [`DropReason::ALL`], shared by every flushed batch.
30    reason_dict: ArrayRef,
31}
32
33impl DroppedReadBuffer {
34    /// Creates a buffer pre-sized to `capacity` rows.
35    pub fn new(capacity: usize) -> Self {
36        Self {
37            read_name: StringColumn::with_capacity(capacity),
38            sequence: StringColumn::with_capacity(capacity),
39            quality: StringColumn::with_capacity(capacity),
40            sequence_2: StringColumn::with_capacity(capacity),
41            quality_2: StringColumn::with_capacity(capacity),
42            reason: Vec::with_capacity(capacity),
43            reason_dict: Arc::new(StringArray::from_iter_values(
44                DropReason::ALL.iter().map(DropReason::as_str),
45            )),
46        }
47    }
48
49    pub fn push<R: Record>(&mut self, read: Read<'_, R>, reason: DropReason) {
50        let (first, mate) = match read {
51            Read::SingleForward(r) | Read::SingleReverse(r) => (r, None),
52            Read::Paired { r1, r2 } => (r1, Some(r2)),
53        };
54        self.read_name.push(Some(read_name(first.id())));
55        self.sequence.push(Some(first.seq().as_ref()));
56        self.quality.push(first.qual());
57        self.sequence_2.push(mate.map(|r| r.seq()).as_deref());
58        self.quality_2.push(mate.and_then(|r| r.qual()));
59        self.reason.push(reason.index() as i32);
60    }
61
62    pub(crate) fn is_empty(&self) -> bool {
63        self.reason.is_empty()
64    }
65
66    pub(crate) fn len(&self) -> usize {
67        self.reason.len()
68    }
69
70    pub(crate) fn schema() -> Arc<Schema> {
71        Arc::new(Schema::new(vec![
72            Field::new("read_name", DataType::Utf8, false),
73            Field::new("sequence", DataType::Utf8, false),
74            // Nullable: paraseq also reads FASTA, which has no qualities.
75            Field::new("quality", DataType::Utf8, true),
76            Field::new("sequence_2", DataType::Utf8, true),
77            Field::new("quality_2", DataType::Utf8, true),
78            Field::new(
79                "reason",
80                DataType::Dictionary(Box::new(DataType::Int32), Box::new(DataType::Utf8)),
81                false,
82            ),
83        ]))
84    }
85
86    /// Moves the buffered rows into an Arrow `RecordBatch` and resets the buffer.
87    ///
88    /// Fails if a read name, sequence or quality string is not valid UTF-8.
89    pub fn flush(&mut self) -> Result<RecordBatch, ArrowError> {
90        let reason = DictionaryArray::<Int32Type>::try_new(
91            Int32Array::from(std::mem::take(&mut self.reason)),
92            self.reason_dict.clone(),
93        )?;
94        let columns: Vec<ArrayRef> = vec![
95            Arc::new(self.read_name.finish()?),
96            Arc::new(self.sequence.finish()?),
97            Arc::new(self.quality.finish()?),
98            Arc::new(self.sequence_2.finish()?),
99            Arc::new(self.quality_2.finish()?),
100            Arc::new(reason),
101        ];
102        RecordBatch::try_new(Self::schema(), columns)
103    }
104}

106/// The read name is the header up to its first whitespace, as samtools and most
107/// tools define it; the rest is a free-form comment (and differs between mates).
108fn read_name(id: &[u8]) -> &[u8] {
109    id.split(u8::is_ascii_whitespace).next().unwrap_or(id)
110}

112/// A growable Arrow `Utf8` column.
113#[derive(Clone)]
114struct StringColumn {
115    values: Vec<u8>,
116    /// `offsets[i]..offsets[i + 1]` delimits row `i` in `values`; always starts
117    /// with a leading 0, so it holds one more entry than there are rows.
118    offsets: Vec<i32>,
119    validity: Vec<bool>,
120    has_nulls: bool,
121}
122
123impl StringColumn {
124    fn with_capacity(rows: usize) -> Self {
125        let mut offsets = Vec::with_capacity(rows + 1);
126        offsets.push(0);
127        Self {
128            values: Vec::new(),
129            offsets,
130            validity: Vec::with_capacity(rows),
131            has_nulls: false,
132        }
133    }
134
135    fn push(&mut self, value: Option<&[u8]>) {
136        if let Some(bytes) = value {
137            self.values.extend_from_slice(bytes);
138        }
139        self.has_nulls |= value.is_none();
140        self.validity.push(value.is_some());
141        // An `i32` offset caps a batch at 2 GiB of text; flushing every
142        // `DROPPED_BATCH_THRESHOLD` rows keeps short reads far below that, and
143        // overflow is still caught by the offset validation in `finish`.
144        self.offsets.push(self.values.len() as i32);
145    }
146
147    /// Hands the buffered values to Arrow (zero-copy) and resets the column.
148    fn finish(&mut self) -> Result<StringArray, ArrowError> {
149        let rows = self.validity.len();
150        let values = std::mem::take(&mut self.values);
151        let offsets = std::mem::replace(&mut self.offsets, {
152            let mut fresh = Vec::with_capacity(rows + 1);
153            fresh.push(0);
154            fresh
155        });
156        let validity = std::mem::replace(&mut self.validity, Vec::with_capacity(rows));
157        let nulls = std::mem::take(&mut self.has_nulls).then(|| NullBuffer::from(validity));
158
159        StringArray::try_new(
160            OffsetBuffer::new(ScalarBuffer::from(offsets)),
161            Buffer::from_vec(values),
162            nulls,
163        )
164    }
165}

174    fn record(id: &str, seq: &str, qual: &str) -> OwnedRecord {
175        OwnedRecord {
176            id: id.as_bytes().to_vec(),
177            seq: seq.as_bytes().to_vec(),
178            qual: qual.as_bytes().to_vec(),

182    fn string_column<'a>(batch: &'a RecordBatch, name: &str) -> &'a StringArray {
183        batch
184            .column_by_name(name)
185            .expect("column present")
186            .as_any()
187            .downcast_ref::<StringArray>()
188            .expect("utf8 column")
189    }
190
191    fn reasons(batch: &RecordBatch) -> Vec<String> {
192        let dict = batch
193            .column_by_name("reason")
194            .expect("reason column")
195            .as_any()
196            .downcast_ref::<DictionaryArray<Int32Type>>()
197            .expect("dictionary-encoded reason");
198        let names = dict
199            .values()
200            .as_any()
201            .downcast_ref::<StringArray>()
202            .expect("string dictionary values");
203        dict.keys()
204            .iter()
205            .map(|k| names.value(k.expect("non-null key") as usize).to_string())
206            .collect()
207    }

209    #[test]
210    fn single_end_read_leaves_mate_columns_null() {
211        let r = record("read0 1:N:0:ACGT", "ACGTN", "IIII#");
212        let mut buf = DroppedReadBuffer::new(4);
213        buf.push(Read::SingleForward(&&r), DropReason::HasN);
214
215        let batch = buf.flush().expect("flush");
216        assert_eq!(batch.schema(), DroppedReadBuffer::schema());
217        assert_eq!(batch.num_rows(), 1);
218        assert_eq!(string_column(&batch, "read_name").value(0), "read0");
219        assert_eq!(string_column(&batch, "sequence").value(0), "ACGTN");
220        assert_eq!(string_column(&batch, "quality").value(0), "IIII#");
221        assert!(string_column(&batch, "sequence_2").is_null(0));
222        assert!(string_column(&batch, "quality_2").is_null(0));
223        assert_eq!(reasons(&batch), ["has_n"]);
224    }

226    #[test]
227    fn paired_read_keeps_r1_name_and_both_mates() {
228        let r1 = record("pair0 1:N:0", "AAAA", "FFFF");
229        let r2 = record("pair0 2:N:0", "CCCCC", "#####");
230        let mut buf = DroppedReadBuffer::new(4);
231        buf.push(
232            Read::Paired { r1: &&r1, r2: &&r2 },
233            DropReason::Pid2NotInPanel,
234        );
235
236        let batch = buf.flush().expect("flush");
237        assert_eq!(string_column(&batch, "read_name").value(0), "pair0");
238        assert_eq!(string_column(&batch, "sequence").value(0), "AAAA");
239        assert_eq!(string_column(&batch, "quality").value(0), "FFFF");
240        assert_eq!(string_column(&batch, "sequence_2").value(0), "CCCCC");
241        assert_eq!(string_column(&batch, "quality_2").value(0), "#####");
242        assert_eq!(reasons(&batch), ["pid2_not_in_panel"]);
243    }

245    #[test]
246    fn flush_resets_the_buffer_for_the_next_batch() {
247        let a = record("a", "AC", "II");
248        let b = record("b", "GTT", "III");
249        let mut buf = DroppedReadBuffer::new(1);
250
251        buf.push(Read::SingleReverse(&&a), DropReason::TooShort);
252        buf.push(Read::SingleReverse(&&b), DropReason::LowQuality);
253        assert_eq!(buf.len(), 2);
254        let first = buf.flush().expect("first flush");
255        assert!(buf.is_empty());
256
257        buf.push(Read::SingleReverse(&&b), DropReason::LbsOverlap);
258        let second = buf.flush().expect("second flush");
259
260        let names = string_column(&first, "read_name");
261        assert_eq!((names.value(0), names.value(1)), ("a", "b"));
262        assert_eq!(string_column(&first, "sequence").value(1), "GTT");
263        assert_eq!(reasons(&first), ["too_short", "low_quality"]);
264
265        assert_eq!(second.num_rows(), 1);
266        assert_eq!(string_column(&second, "sequence").value(0), "GTT");
267        assert_eq!(reasons(&second), ["lbs_overlap"]);
268    }
