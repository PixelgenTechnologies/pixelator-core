11#[derive(Clone)]
12pub struct RowBuffer {
13    /// Marker indices for `pid-1`; each entry is a key into `marker_dict` and is
14    /// emitted as a dictionary-encoded (VARCHAR) `marker_1` column, and into
15    /// `seq1_dict` for the `marker_sequence_1` column.
16    buf_marker_idx1: Vec<i32>,
17    /// Marker indices for `pid-2`; each entry is a key into `marker_dict` and is
18    /// emitted as a dictionary-encoded (VARCHAR) `marker_2` column, and into
19    /// `seq2_dict` for the `marker_sequence_2` column.
20    buf_marker_idx2: Vec<i32>,
21    buf_umi1: Vec<u64>,
22    buf_umi2: Vec<u64>,
23    /// Shared dictionary values: the panel's marker ids in file order, indexed
24    /// by the `buf_marker_idx*` keys. Built once in `new` and reused (as a cheap
25    /// `Arc` clone) for every flushed batch.
26    marker_dict: ArrayRef,
27    /// Likewise the panel's `sequence_1` barcodes, indexed by `buf_marker_idx1`.
28    seq1_dict: ArrayRef,
29    /// Likewise the panel's `sequence_2` barcodes, indexed by `buf_marker_idx2`.
30    seq2_dict: ArrayRef,
31}
32
33impl RowBuffer {
34    /// Creates a buffer pre-sized to `capacity` rows.
35    ///
36    /// The `panel` is only borrowed to snapshot its marker ids and barcode
37    /// sequences into the shared dictionary values arrays. The marker indices later pushed via
38    /// [`push`](Self::push) must be positions in this same panel's `markers`
39    /// (which is what [`Panel::marker_index_seq1`] and friends return).
40    pub fn new(capacity: usize, panel: &Panel) -> Self {
41        let marker_dict: ArrayRef = Arc::new(StringArray::from_iter_values(
42            panel.markers().iter().map(|m| m.marker_id.as_str()),
43        ));
44        let seq1_dict: ArrayRef = Arc::new(StringArray::from_iter_values(
45            panel.markers().iter().map(|m| m.sequence_1.as_str()),
46        ));
47        let seq2_dict: ArrayRef = Arc::new(StringArray::from_iter_values(
48            panel.markers().iter().map(|m| m.sequence_2.as_str()),
49        ));
50
51        Self {
52            buf_marker_idx1: Vec::with_capacity(capacity),
53            buf_marker_idx2: Vec::with_capacity(capacity),
54            buf_umi1: Vec::with_capacity(capacity),
55            buf_umi2: Vec::with_capacity(capacity),
56            marker_dict,
57            seq1_dict,
58            seq2_dict,
59        }
60    }
61
62    pub fn push(&mut self, marker_idx1: i32, marker_idx2: i32, umi1: u64, umi2: u64) {
63        self.buf_marker_idx1.push(marker_idx1);
64        self.buf_marker_idx2.push(marker_idx2);
65        self.buf_umi1.push(umi1);
66        self.buf_umi2.push(umi2);
67    }
68
69    pub(crate) fn is_empty(&self) -> bool {
70        self.buf_marker_idx1.len() == 0
71    }
72
73    pub(crate) fn len(&self) -> usize {
74        self.buf_marker_idx1.len()
75    }
76
77    pub(crate) fn schema() -> Arc<Schema> {
78        Arc::new(Schema::new(vec![
79            Field::new(
80                "marker_1",
81                DataType::Dictionary(Box::new(DataType::Int32), Box::new(DataType::Utf8)),
82                false,
83            ),
84            Field::new(
85                "marker_2",
86                DataType::Dictionary(Box::new(DataType::Int32), Box::new(DataType::Utf8)),
87                false,
88            ),
89            Field::new("umi1", DataType::UInt64, false),
90            Field::new("umi2", DataType::UInt64, false),
91            Field::new(
92                "marker_sequence_1",
93                DataType::Dictionary(Box::new(DataType::Int32), Box::new(DataType::Utf8)),
94                false,
95            ),
96            Field::new(
97                "marker_sequence_2",
98                DataType::Dictionary(Box::new(DataType::Int32), Box::new(DataType::Utf8)),
99                false,
100            ),
101        ]))
102    }
103
104    /// Move the buffered columns into an Arrow `RecordBatch` and reset the buffers, preserving each
105    /// vector's previous capacity for the next batch.
106    ///
107    /// The `Vec`s are handed to Arrow zero-copy: `Buffer::from_vec` adopts each vector's existing
108    /// allocation rather than copying elements, so both the `i32` marker keys and the `u64` UMIs
109    /// are transferred without a per-row copy.
110    ///
111    /// The four dictionary columns are built as `DictionaryArray`s over `Arc`'d dictionaries
112    /// created once in `new`: the marker ids, and the panel's `sequence_1` and `sequence_2`
113    /// barcodes. Each side's `Int32Array` of keys is built once and shared between its marker
114    /// column and its sequence column, as the sequence is looked up by the same marker index.
115    /// Cloning a dictionary or a key array is just a refcount bump.
116    ///
117    /// Parquet stores the dictionary columns as RLE_DICTIONARY-encoded VARCHAR. The reset leaves
118    /// freshly allocated buffers, pre-sized to avoid re-growing while filling the next batch.
119    pub fn flush(&mut self) -> Result<RecordBatch, ArrowError> {
120        let keys1 = Int32Array::from(take_buffer(&mut self.buf_marker_idx1));
121        let keys2 = Int32Array::from(take_buffer(&mut self.buf_marker_idx2));
122        let umi1 = UInt64Array::from(take_buffer(&mut self.buf_umi1));
123        let umi2 = UInt64Array::from(take_buffer(&mut self.buf_umi2));
124
125        let marker1 =
126            DictionaryArray::<Int32Type>::try_new(keys1.clone(), self.marker_dict.clone())?;
127        let marker2 =
128            DictionaryArray::<Int32Type>::try_new(keys2.clone(), self.marker_dict.clone())?;
129        let seq1 = DictionaryArray::<Int32Type>::try_new(keys1, self.seq1_dict.clone())?;
130        let seq2 = DictionaryArray::<Int32Type>::try_new(keys2, self.seq2_dict.clone())?;
131
132        let columns: Vec<ArrayRef> = vec![
133            Arc::new(marker1),
134            Arc::new(marker2),
135            Arc::new(umi1),
136            Arc::new(umi2),
137            Arc::new(seq1),
138            Arc::new(seq2),
139        ];
140
141        RecordBatch::try_new(Self::schema(), columns)
142    }
143}
144
145/// Reset a column buffer, returning its contents and leaving a fresh buffer
146/// with the same capacity for the next batch.
147fn take_buffer<T>(buf: &mut Vec<T>) -> Vec<T> {
148    let cap = buf.capacity();
149    mem::replace(buf, Vec::with_capacity(cap))
150}

160    /// Builds the expected `DictionaryArray` for a column from a set of keys and
161    /// the panel used to construct the buffer; `value` picks which marker field
162    /// (id or one of the sequences) the dictionary holds.
163    fn expected_dict_column(
164        keys: Vec<i32>,
165        panel: &Panel,
166        value: fn(&PanelMarker) -> &str,
167    ) -> DictionaryArray<Int32Type> {
168        let values: ArrayRef = Arc::new(StringArray::from_iter_values(
169            panel.markers().iter().map(value),
170        ));
171        DictionaryArray::<Int32Type>::try_new(Int32Array::from(keys), values).unwrap()
172    }

174    /// Resolves a dictionary column back to its per-row strings.
175    fn column_strings(batch: &RecordBatch, name: &str) -> Vec<String> {
176        let column = batch.column_by_name(name).expect("column present");
177        let dict = column
178            .as_any()
179            .downcast_ref::<DictionaryArray<Int32Type>>()
180            .expect("dictionary-encoded column")
181            .downcast_dict::<StringArray>()
182            .expect("string dictionary values");
183        (0..dict.len()).map(|i| dict.value(i).to_string()).collect()
184    }

188        let cap = 128;
189        let rb = RowBuffer::new(cap, &proxiome_v2_panel());
190
191        // `with_capacity` guarantees "at least cap", so assert the lower bound.
192        assert!(rb.buf_marker_idx1.capacity() >= cap);
193        assert!(rb.buf_marker_idx2.capacity() >= cap);
194        assert!(rb.buf_umi1.capacity() >= cap);
195        assert!(rb.buf_umi2.capacity() >= cap);

249    /// Each sequence column resolves to the panel barcode of the marker in the
250    /// matching marker column: `sequence_1` for side 1, `sequence_2` for side 2.
251    #[test]
252    fn flush_sequences_match_the_panel_barcodes_of_their_markers() {
253        let panel = proxiome_v2_panel();
254        let mut rb = RowBuffer::new(4, &panel);
255        let rows = [(0, 2), (3, 1), (1, 1)];
256        for (idx1, idx2) in rows {
257            rb.push(idx1, idx2, 0, 0);
258        }
259
260        let batch = rb.flush().expect("flush");
261
262        let markers = panel.markers();
263        let expect = |idx: fn(&(i32, i32)) -> i32, field: fn(&PanelMarker) -> &str| {
264            rows.iter()
265                .map(|row| field(&markers[idx(row) as usize]).to_string())
266                .collect::<Vec<_>>()
267        };
268        assert_eq!(
269            column_strings(&batch, "marker_1"),
270            expect(|r| r.0, |m| m.marker_id.as_str())
271        );
272        assert_eq!(
273            column_strings(&batch, "marker_2"),
274            expect(|r| r.1, |m| m.marker_id.as_str())
275        );
276        assert_eq!(
277            column_strings(&batch, "marker_sequence_1"),
278            expect(|r| r.0, |m| m.sequence_1.as_str())
279        );
280        assert_eq!(
281            column_strings(&batch, "marker_sequence_2"),
282            expect(|r| r.1, |m| m.sequence_2.as_str())

303    fn flush_does_not_carry_over_previous_rows() {
304        let panel = proxiome_v2_panel();

306        rb.push(5, 1, 11, 21);
