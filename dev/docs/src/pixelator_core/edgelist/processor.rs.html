18// Tracked on its own rather than derived from EdgelistStats, which already counts
19// every read. The two have different lifecycles: a worker's stats are folded into
20// the global tally and reset at end of thread, so reading progress off them couples
21// the two and turns that reset into a hazard.
22//
23// The shared counter is cumulative and is never reset, unlike the per-thread
24// tallies it runs alongside.
25
26/// Publishes read counts into a counter shared with a progress reporter,
27/// batching the atomic writes so the per-read path stays thread-local.
28#[derive(Clone, Default)]
29struct Progress {
30    shared: Arc<AtomicU64>,
31    /// Reads counted since the last [`Progress::publish`].
32    pending: u64,
33}
34
35impl Progress {
36    fn new(shared: Arc<AtomicU64>) -> Self {
37        Self { shared, pending: 0 }
38    }
39
40    /// Counts one processed read, thread-locally.
41    fn count_read(&mut self) {
42        self.pending += 1;
43    }
44
45    /// Publishes and clears whatever has accumulated since the last call.
46    ///
47    /// Called from the batch/thread-complete hooks, never from the per-read hot
48    /// path, so a worker touches the shared atomic once per batch rather than
49    /// once per read.
50    fn publish(&mut self) {
51        let pending = std::mem::take(&mut self.pending);
52        if pending > 0 {
53            self.shared.fetch_add(pending, Ordering::Relaxed);
54        }
55    }
56}

58/// Where the reads the extractor discards go when the run saves them: a buffer
59/// flushed to its own channel, so the dropped-reads Parquet is written by a
60/// writer thread of its own alongside the edgelist's.
61#[derive(Clone)]
62struct DroppedReadSink {
63    buffer: DroppedReadBuffer,
64    tx: SyncSender<RecordBatch>,
65    batch_threshold: usize,
66}
67
68impl DroppedReadSink {
69    fn new(tx: SyncSender<RecordBatch>, batch_threshold: usize) -> Self {
70        Self {
71            buffer: DroppedReadBuffer::new(batch_threshold),
72            tx,
73            batch_threshold,
74        }
75    }
76
77    fn flush_and_send(&mut self) -> Result<(), ProcessError> {
78        let batch = self
79            .buffer
80            .flush()
81            .map_err(IntoProcessError::into_process_error)?;
82        self.tx
83            .send(batch)
84            .map_err(IntoProcessError::into_process_error)
85    }
86}

88/// The mode-agnostic core shared by [`PairedProcessor`] and [`SingleProcessor`].
89///
90/// Owns the batch machinery (row buffer, channel, stats, extractor) and all the
91/// logic that does not depend on how a [`Read`] is assembled, so the per-mode
92/// processors stay thin wrappers that only decide which `Read` variant to build.
93#[derive(Clone)]
94struct RowSink {
95    buffer: RowBuffer,
96
97    tx: SyncSender<RecordBatch>,
98    batch_threshold: usize,
99
100    local_stats: EdgelistStats,
101    global_stats: Arc<Mutex<EdgelistStats>>,
102
103    progress: Progress,
104
105    extractor: Extractor,
106    qc_profiler: QcProfiler,
107
108    /// `None` unless the run saves dropped reads, so the default path pays
109    /// nothing for the feature.
110    dropped: Option<DroppedReadSink>,
111}
112
113impl RowSink {
114    fn new(
115        tx: SyncSender<RecordBatch>,
116        batch_threshold: usize,
117        assay: Assay,
118        panel: Panel,
119        filters: FilterConfig,
120    ) -> Self {
121        Self {
122            buffer: RowBuffer::new(batch_threshold, &panel),
123            tx,
124            batch_threshold,
125            local_stats: EdgelistStats::default(),
126            global_stats: Arc::new(Mutex::new(EdgelistStats::default())),
127            progress: Progress::default(),
128            extractor: Extractor::new(&assay, panel, filters),
129            qc_profiler: QcProfiler::new(&assay),
130            dropped: None,
131        }
132    }
133
134    /// Extracts one molecule from `read` and records the outcome: a valid row is
135    /// buffered and counted as kept; a [`DropReason`] is tallied, and the read
136    /// itself buffered when dropped reads are being saved.
137    ///
138    /// QC is measured first, ahead of the keep/drop decision, so that dropped
139    /// reads still contribute to the quality totals.
140    fn process<Rf: Record>(&mut self, read: Read<'_, Rf>) {
141        self.progress.count_read();
142        self.local_stats.add_qc(self.qc_profiler.eval(&read));
143        match self.extractor.extract(&read) {
144            Ok(row) => {
145                self.buffer
146                    .push(row.marker_idx1, row.marker_idx2, row.umi1, row.umi2);
147                self.local_stats.keep_read(row.marker_idx1, row.marker_idx2);
148            }
149            Err(reason) => self.drop_read(read, reason),
150        }
151    }
152
153    fn drop_read<Rf: Record>(&mut self, read: Read<'_, Rf>, reason: DropReason) {
154        self.local_stats.drop_read(reason);
155        if let Some(dropped) = &mut self.dropped {
156            dropped.buffer.push(read, reason);
157        }
158    }
159
160    fn flush_and_send(&mut self) -> Result<(), ProcessError> {
161        let batch = self
162            .buffer
163            .flush()
164            .map_err(IntoProcessError::into_process_error)?;
165        self.tx
166            .send(batch)
167            .map_err(IntoProcessError::into_process_error)
168    }
169
170    /// Batch boundary (`on_batch_complete`): publishes progress and flushes each
171    /// buffer that has reached its batch threshold.
172    fn finish_batch(&mut self) -> Result<(), ProcessError> {
173        self.progress.publish();
174        if self.buffer.len() >= self.batch_threshold {
175            self.flush_and_send()?;
176        }
177        if let Some(dropped) = &mut self.dropped
178            && dropped.buffer.len() >= dropped.batch_threshold
179        {
180            dropped.flush_and_send()?;
181        }
182        Ok(())
183    }
184
185    /// End of thread (`on_thread_complete`): publishes the last of the progress,
186    /// flushes any trailing rows (kept and dropped), and folds this worker's
187    /// stats into the shared tally.
188    fn finish_thread(&mut self) -> Result<(), ProcessError> {
189        self.progress.publish();
190        if !self.buffer.is_empty() {
191            self.flush_and_send()?;
192        }
193        if let Some(dropped) = &mut self.dropped
194            && !dropped.buffer.is_empty()
195        {
196            dropped.flush_and_send()?;
197        }
198
199        let mut global_stats = self.global_stats.lock().expect("lock global stats");
200        *global_stats += std::mem::take(&mut self.local_stats);
201
202        Ok(())
203    }
204}

206/// Paired-end processor: side one comes from the forward read, side two from the
207/// reverse read.
208#[derive(Clone)]
209pub struct PairedProcessor {
210    sink: RowSink,
211}
212
213impl PairedProcessor {
214    pub fn new(
215        tx: SyncSender<RecordBatch>,
216        batch_threshold: usize,
217        assay: Assay,
218        panel: Panel,
219        filters: FilterConfig,
220    ) -> Self {
221        Self {
222            sink: RowSink::new(tx, batch_threshold, assay, panel, filters),
223        }
224    }
225
226    /// Share `counter` across worker clones so a reporter can observe progress.
227    pub fn with_progress_counter(mut self, counter: Arc<AtomicU64>) -> Self {
228        self.sink.progress = Progress::new(counter);
229        self
230    }
231
232    /// Also send every dropped read, with its reason, down `tx` in batches of
233    /// about `batch_threshold` rows (see [`DroppedReadBuffer`] for the schema).
234    /// `None` leaves dropped reads unsaved, as without this call.
235    pub fn with_dropped_reads(
236        mut self,
237        tx: Option<SyncSender<RecordBatch>>,
238        batch_threshold: usize,
239    ) -> Self {
240        self.sink.dropped = tx.map(|tx| DroppedReadSink::new(tx, batch_threshold));
241        self
242    }
243
244    /// Consumes the processor and returns the tally merged from every worker.
245    ///
246    /// The counts only become final once `process_parallel` has returned, since
247    /// each worker folds its local tally in as it finishes.
248    pub fn into_stats(self) -> EdgelistStats {
249        // Taking `self` enforces that ordering: `process_parallel` borrows the
250        // processor mutably, so reading the tally and then resuming the run does
251        // not compile.
252        self.sink
253            .global_stats
254            .lock()
255            .expect("lock global stats")
256            .clone()
257    }
258}
259
260impl<Rf: Record> PairedParallelProcessor<Rf> for PairedProcessor {
261    fn process_record_pair(&mut self, r1: Rf, r2: Rf) -> Result<(), ProcessError> {
262        self.sink.process(Read::Paired { r1: &r1, r2: &r2 });
263        Ok(())
264    }
265
266    fn on_batch_complete(&mut self) -> Result<(), ProcessError> {
267        self.sink.finish_batch()
268    }
269
270    fn on_thread_complete(&mut self) -> Result<(), ProcessError> {
271        self.sink.finish_thread()
272    }
273}

721    /// At end of thread the worker's local stats — QC included — are folded into
722    /// the shared global stats, and the local ones are reset.
723    #[test]
724    fn finish_thread_folds_qc_into_global_stats() {
725        use crate::test_fixtures::{
726            ReadMixConfig, generate_read_pairs, proxiome_v2_marker_sequences,
727        };
728        use paraseq::parallel::PairedParallelProcessor;
729
730        let assay = proxiome_v2_assay();
731        let markers = proxiome_v2_marker_sequences();
732        let mix = ReadMixConfig {
733            frac_has_n: 0.0,
734            frac_unknown_pid: 0.0,
735            read_len: 150,
736        };
737        let records = generate_read_pairs(&assay, &markers, 50, mix, 5);
738
739        let (mut proc, _rx) = make_processor(records.len());
740        for (r1, r2) in &records {
741            proc.process_record_pair(r1, r2)
742                .expect("process record pair");
743        }
744
745        let local_qc = proc.sink.local_stats.read_qc.clone();
746        assert_ne!(

748            "reads should have been measured before the flush"
749        );

751        proc.sink.finish_thread().expect("thread complete");
752

759            local_qc.q30_whole.total_bases

762            global.read_qc.q30_pid_a.total_bases,
763            local_qc.q30_pid_a.total_bases
764        );
765        assert_eq!(global.total_kept_reads(), records.len());
766    }

768    /// `edgelist` calls this on the one handle it kept, but the tally it needs
769    /// covers every worker paraseq cloned from that handle. Both clones fold
770    /// their own reads in, and the returned value has to show both.
771    #[test]
772    fn into_stats_returns_the_tally_folded_by_every_worker() {
773        use crate::test_fixtures::{
774            ReadMixConfig, generate_read_pairs, proxiome_v2_marker_sequences,
775        };
776        use paraseq::parallel::PairedParallelProcessor;
777
778        let assay = proxiome_v2_assay();
779        let markers = proxiome_v2_marker_sequences();
780        let mix = ReadMixConfig {
781            frac_has_n: 0.0,
782            frac_unknown_pid: 0.0,
783            read_len: 150,
784        };
785        let records = generate_read_pairs(&assay, &markers, 20, mix, 7);
786
787        let (mut proc, rx) = make_processor(2 * records.len());
788        let mut worker = proc.clone();
789        for (r1, r2) in &records {
790            proc.process_record_pair(r1, r2)
791                .expect("process record pair");
792            worker
793                .process_record_pair(r1, r2)
794                .expect("process record pair");
795        }
796
797        proc.sink.finish_thread().expect("thread complete");
798        // The channel buffers a single batch, so drain the first worker's
799        // before the second one flushes or the send blocks forever.
800        rx.recv().expect("first worker's batch");
801        worker.sink.finish_thread().expect("worker thread complete");
802
803        let stats = proc.into_stats();
804
805        assert_eq!(stats.total_kept_reads(), 2 * records.len());
806        assert_ne!(
807            stats.read_qc.q30_whole.total_bases, 0,
808            "QC should fold in alongside the read counts"
809        );
810    }
