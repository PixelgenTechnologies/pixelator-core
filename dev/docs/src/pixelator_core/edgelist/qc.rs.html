1//! Quality metrics measured over raw reads, whatever the pipeline does with them.
2//!
3//! Everything here measures rather than decides. A metric covers every read the
4//! run sees, including the ones [`extract`](super::extract) goes on to discard,
5//! and computing one never fails: a read too short to reach a region contributes
6//! nothing to that region's tally instead of reporting an error.
7
8// A metric belongs here when it can be read off the raw bases alone. One that
9// depends on what extraction decided does not - per-marker read counts need a
10// [`Row`](super::extract::Row), so they are tallied by the processor, where that
11// outcome is known.
12

19/// Phred+33 byte above which a base is "Q30" (quality >= 30)
20const Q30_MIN_BYTE: u8 = 33 + 30; // 63
21
22/// A Q30 tally over some set of bases: how many reached Q30, out of how many
23/// were examined.
24///
25/// Both halves are kept rather than just their ratio so that tallies from
26/// different reads can be summed before [`Q30Count::rate`] is taken. Averaging
27/// per-read rates instead would give a short read the same weight as a long
28/// one.
29#[derive(Default, Clone, Copy)]
30pub struct Q30Count {
31    pub q30_bases: usize,
32    pub total_bases: usize,
33}
34
35impl Q30Count {
36    fn count(qual: &[u8]) -> Self {
37        let q30_bases = qual.iter().filter(|&&b| b >= Q30_MIN_BYTE).count();
38        Self {
39            q30_bases,
40            total_bases: qual.len(),
41        }
42    }
43
44    pub fn rate(&self) -> f64 {
45        if self.total_bases == 0 {
46            0.0
47        } else {
48            self.q30_bases as f64 / self.total_bases as f64
49        }
50    }
51}
52
53impl Add for Q30Count {
54    type Output = Self;
55
56    fn add(self, other: Self) -> Self {
57        Self {
58            q30_bases: self.q30_bases + other.q30_bases,
59            total_bases: self.total_bases + other.total_bases,
60        }
61    }
62}
63
64impl AddAssign for Q30Count {
65    fn add_assign(&mut self, other: Self) {
66        self.q30_bases += other.q30_bases;
67        self.total_bases += other.total_bases;
68    }
69}

71/// The Q30 profile of one read or read pair, as produced by
72/// [`QcProfiler::eval`].
73///
74/// `q30_whole` covers every base that was sequenced, both mates included for a
75/// pair, while the four region tallies cover only the `pid`/`umi` bases of each
76/// conjugate. They are named after [`ConjugateSide`] rather than the assay's
77/// `pid-1`/`pid-2` region ids, since which read a conjugate is read from
78/// depends on the sequencing mode.
79#[derive(Default, Clone)]
80pub struct ReadQc {
81    pub q30_whole: Q30Count,
82    pub q30_pid_a: Q30Count,
83    pub q30_umi_a: Q30Count,
84    pub q30_pid_b: Q30Count,
85    pub q30_umi_b: Q30Count,
86}
87
88impl AddAssign for ReadQc {
89    fn add_assign(&mut self, other: Self) {
90        self.q30_whole += other.q30_whole;
91        self.q30_pid_a += other.q30_pid_a;
92        self.q30_umi_a += other.q30_umi_a;
93        self.q30_pid_b += other.q30_pid_b;
94        self.q30_umi_b += other.q30_umi_b;
95    }
96}

98/// Evaluates Q30 for a specific region of the quality string.
99fn region_counts(qual: Option<&[u8]>, (s, e): (usize, usize)) -> Q30Count {
100    match qual.and_then(|q| q.get(s..e)) {
101        Some(slice) => Q30Count::count(slice),
102        None => Q30Count::default(),
103    }
104}

106/// Measures the Q30 profile of a read: the whole read, plus one tally per
107/// region of interest.
108///
109/// Region offsets are resolved from the assay once, at construction, into an
110/// owned [`RegionLayout`].
111#[derive(Clone)]
112pub struct QcProfiler {
113    layout: RegionLayout,
114}
115
116impl QcProfiler {
117    pub fn new(assay: &Assay) -> Self {
118        Self {
119            layout: RegionLayout::new(assay),
120        }
121    }
122
123    /// Measures `read`, reading each conjugate's regions at the bounds for the
124    /// orientation it was sequenced in.
125    ///
126    /// This always succeeds. A region the read is too short to reach simply
127    /// contributes nothing to its tally, so there is no failure to report and
128    /// no read that goes unmeasured.
129    pub fn eval<R: Record>(&self, read: &Read<'_, R>) -> ReadQc {
130        let q30_whole = |r: &R| r.qual().map(Q30Count::count).unwrap_or_default();
131
132        match read {
133            Read::SingleForward(r) => {
134                let bounds_a = self.layout.get(ConjugateSide::A, Strand::Forward);
135                let bounds_b = self.layout.get(ConjugateSide::B, Strand::Forward);
136                ReadQc {
137                    q30_whole: q30_whole(r),
138                    q30_pid_a: region_counts(r.qual(), bounds_a.pid),
139                    q30_umi_a: region_counts(r.qual(), bounds_a.umi),
140                    q30_pid_b: region_counts(r.qual(), bounds_b.pid),
141                    q30_umi_b: region_counts(r.qual(), bounds_b.umi),
142                }
143            }
144            Read::SingleReverse(r) => {
145                let bounds_a = self.layout.get(ConjugateSide::A, Strand::Reverse);
146                let bounds_b = self.layout.get(ConjugateSide::B, Strand::Reverse);
147                ReadQc {
148                    q30_whole: q30_whole(r),
149                    q30_pid_a: region_counts(r.qual(), bounds_a.pid),
150                    q30_umi_a: region_counts(r.qual(), bounds_a.umi),
151                    q30_pid_b: region_counts(r.qual(), bounds_b.pid),
152                    q30_umi_b: region_counts(r.qual(), bounds_b.umi),
153                }
154            }
155            Read::Paired { r1, r2 } => {
156                let bounds_a = self.layout.get(ConjugateSide::A, Strand::Forward);
157                let bounds_b = self.layout.get(ConjugateSide::B, Strand::Reverse);
158                ReadQc {
159                    q30_whole: q30_whole(r1) + q30_whole(r2),
160                    q30_pid_a: region_counts(r1.qual(), bounds_a.pid),
161                    q30_umi_a: region_counts(r1.qual(), bounds_a.umi),
162                    q30_pid_b: region_counts(r2.qual(), bounds_b.pid),
163                    q30_umi_b: region_counts(r2.qual(), bounds_b.umi),
164                }
165            }
166        }
167    }
168}

178    /// Phred+33 byte well below the threshold (Q10).
179    const LOW: u8 = b'+';
180    /// Phred+33 byte well above the threshold (Q40).
181    const HIGH: u8 = b'I';
182
183    /// Number of Q40 bases to plant in each region of interest.
184    type Fingerprint = [(&'static str, usize); 4];
185
186    /// `pid-1`/`pid-2` are both 10 bases and `umi-1`/`umi-2` both 28, so region
187    /// lengths alone cannot tell the four apart. Planting a distinct number of
188    /// Q40 bases in each makes a misrouted region (wrong conjugate, wrong
189    /// strand, wrong field) show up as the wrong count. The two fingerprints
190    /// are disjoint so that in paired mode all eight (read, region)
191    /// combinations yield different numbers.
192    const FP_R1: Fingerprint = [("umi-1", 1), ("pid-1", 2), ("pid-2", 3), ("umi-2", 4)];
193    const FP_R2: Fingerprint = [("umi-1", 5), ("pid-1", 6), ("pid-2", 7), ("umi-2", 8)];
194
195    /// An amplicon-length quality string that is Q10 everywhere except for the
196    /// leading `n_high` bases of each fingerprinted region.
197    fn fingerprint_qual(assay: &Assay, strand: Strand, fingerprint: Fingerprint) -> Vec<u8> {
198        let (_, amplicon_len) = assay.forward_bounds("amplicon").expect("amplicon bounds");
199        let mut qual = vec![LOW; amplicon_len];
200        for (region_id, n_high) in fingerprint {
201            let (start, end) = strand.bounds(assay, region_id).expect("region bounds");
202            assert!(start + n_high <= end, "fingerprint must fit in {region_id}");
203            qual[start..start + n_high].fill(HIGH);
204        }
205        qual
206    }
207
208    /// QC only reads the quality string, so the bases themselves are filler.
209    fn record_with_qual(qual: Vec<u8>) -> OwnedRecord {
210        OwnedRecord {
211            id: Vec::new(),
212            seq: vec![b'A'; qual.len()],
213            qual,
214        }
215    }
216
217    fn region_len(assay: &Assay, region_id: &str) -> usize {
218        let (start, end) = assay.forward_bounds(region_id).expect("region bounds");
219        end - start
220    }

318    #[test]
319    fn the_two_orientations_of_one_molecule_give_the_same_counts() {
320        let assay = proxiome_v2_assay();
321        let profiler = QcProfiler::new(&assay);
322
323        // Sequencing the same molecule from the other end reverses the read,
324        // and with it the quality string. Each region then holds the same
325        // bases in the opposite order, which cannot change a count.
326        let forward_qual = fingerprint_qual(&assay, Strand::Forward, FP_R1);
327        let reverse_qual: Vec<u8> = forward_qual.iter().rev().copied().collect();
328        let forward = record_with_qual(forward_qual);
329        let reverse = record_with_qual(reverse_qual);
330        let (forward, reverse) = (&forward, &reverse);
331
332        let fwd = profiler.eval(&Read::SingleForward(&forward));
333        let rev = profiler.eval(&Read::SingleReverse(&reverse));
334
335        assert_eq!(rev.q30_umi_a.q30_bases, fwd.q30_umi_a.q30_bases);
336        assert_eq!(rev.q30_pid_a.q30_bases, fwd.q30_pid_a.q30_bases);
337        assert_eq!(rev.q30_pid_b.q30_bases, fwd.q30_pid_b.q30_bases);
338        assert_eq!(rev.q30_umi_b.q30_bases, fwd.q30_umi_b.q30_bases);
339        assert_eq!(rev.q30_whole.q30_bases, fwd.q30_whole.q30_bases);
340    }

342    #[test]
343    fn paired_takes_conjugate_a_from_r1_and_conjugate_b_from_r2() {
344        let assay = proxiome_v2_assay();
345        let profiler = QcProfiler::new(&assay);
346        let r1 = record_with_qual(fingerprint_qual(&assay, Strand::Forward, FP_R1));
347        let r2 = record_with_qual(fingerprint_qual(&assay, Strand::Reverse, FP_R2));
348        let (r1, r2) = (&r1, &r2);
349
350        let qc = profiler.eval(&Read::Paired { r1: &r1, r2: &r2 });
351
352        // A side comes from R1 read forward, B side from R2 read reverse.
353        assert_eq!(qc.q30_umi_a.q30_bases, 1);
354        assert_eq!(qc.q30_pid_a.q30_bases, 2);
355        assert_eq!(qc.q30_pid_b.q30_bases, 7);
356        assert_eq!(qc.q30_umi_b.q30_bases, 8);
357        // The whole-read counts span both mates.
358        assert_eq!(qc.q30_whole.q30_bases, (1 + 2 + 3 + 4) + (5 + 6 + 7 + 8));
359        assert_eq!(qc.q30_whole.total_bases, r1.qual.len() + r2.qual.len());
360    }

402    #[test]
403    fn generated_pairs_accumulate_over_every_read() {
404        let assay = proxiome_v2_assay();
405        let markers = proxiome_v2_marker_sequences();
406        let mix = ReadMixConfig::default();
407        let records = generate_read_pairs(&assay, &markers, 20, mix, 7);
408        let profiler = QcProfiler::new(&assay);
409
410        let mut total = ReadQc::default();
411        for (r1, r2) in &records {
412            total += profiler.eval(&Read::Paired { r1: &r1, r2: &r2 });
413        }
414
415        // The fixtures emit a constant 'I' (Q40) quality string, and dropped
416        // reads are measured just like kept ones.
417        assert_eq!(total.q30_whole.rate(), 1.0);
418        assert_eq!(total.q30_pid_a.rate(), 1.0);
419        assert_eq!(total.q30_umi_b.rate(), 1.0);
420        assert_eq!(total.q30_whole.total_bases, 20 * 2 * mix.read_len);
421        assert_eq!(
422            total.q30_pid_a.total_bases,
423            20 * region_len(&assay, "pid-1")
424        );
425    }
