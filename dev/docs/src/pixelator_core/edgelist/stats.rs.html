1//! What one edgelist run counted: reads kept (in total and per marker), why the
2//! rest were dropped, and the Q30 profile of everything that came through.
3//!
4//! The layout here serves accumulation, not reporting: values combine with
5//! [`AddAssign`], so every figure is stored in a form that adds, and the
6//! report's fractions and names are derived by the accessors rather than
7//! stored.
8
9use std::cmp::Reverse;
10use std::ops::AddAssign;
11
12use rustc_hash::FxHashMap as HashMap;

19/// Read tallies accumulated over an edgelist run.
20#[derive(Clone, Default)]
21pub struct EdgelistStats {
22    kept: usize,
23    /// Per-reason breakdown of the dropped reads.
24    pub(super) dropped_by: HashMap<DropReason, usize>,
25    pub(super) read_qc: ReadQc,
26    /// Kept reads per marker on each side, keyed by panel position - the same
27    /// index the row buffer stores. Markers never seen have no entry; names are
28    /// attached by the ranking accessors, which take the panel.
29    marker1_reads: HashMap<i32, usize>,
30    marker2_reads: HashMap<i32, usize>,
31}
32
33/// Q30 fractions under the names the report uses.
34///
35/// Each is the share of bases at Phred 30 or better, pooled over every read the
36/// run saw - dropped reads included, since quality is measured ahead of the
37/// keep/drop decision.
38#[derive(Clone, Copy, Debug, Default, PartialEq)]
39pub struct Q30Statistics {
40    /// Whole read: R1 and R2 together when paired, the single read otherwise.
41    pub total: f64,
42    /// Over the pid-1 region.
43    pub pid1: f64,
44    /// Over the mui-1 region.
45    pub umi1: f64,
46    /// Over the pid-2 region.
47    pub pid2: f64,
48    /// Over the umi-2 region.
49    pub umi2: f64,
50}
51
52impl AddAssign for EdgelistStats {
53    fn add_assign(&mut self, other: Self) {
54        self.kept += other.kept;
55        for (reason, count) in other.dropped_by {
56            *self.dropped_by.entry(reason).or_insert(0) += count;
57        }
58        self.read_qc += other.read_qc;
59        add_counts(&mut self.marker1_reads, other.marker1_reads);
60        add_counts(&mut self.marker2_reads, other.marker2_reads);
61    }
62}
63
64fn add_counts(counts: &mut HashMap<i32, usize>, other: HashMap<i32, usize>) {
65    for (marker_idx, count) in other {
66        *counts.entry(marker_idx).or_insert(0) += count;
67    }
68}

70/// Pairs each of `panel`'s marker ids with its tally, most frequent first.
71fn rank_markers<'p>(panel: &'p Panel, counts: &HashMap<i32, usize>) -> Vec<(&'p str, usize)> {
72    let markers = panel.markers();
73    // Walking the panel below would silently skip an index it does not hold.
74    debug_assert!(
75        counts.keys().all(|&idx| (idx as usize) < markers.len()),
76        "marker tallies do not come from this panel"
77    );
78    let mut ranked: Vec<_> = markers
79        .iter()
80        .enumerate()
81        .map(|(idx, marker)| {
82            let count = counts.get(&(idx as i32)).copied().unwrap_or_default();
83            (marker.marker_id.as_str(), count)
84        })
85        .collect();
86    ranked.sort_unstable_by_key(|&(id, count)| (Reverse(count), id));
87    ranked
88}

90impl EdgelistStats {
91    pub(super) fn keep_read(&mut self, marker_idx1: i32, marker_idx2: i32) {
92        self.kept += 1;
93        *self.marker1_reads.entry(marker_idx1).or_insert(0) += 1;
94        *self.marker2_reads.entry(marker_idx2).or_insert(0) += 1;
95    }
96
97    pub(super) fn drop_read(&mut self, reason: DropReason) {
98        *self.dropped_by.entry(reason).or_insert(0) += 1;
99    }
100
101    pub(super) fn add_qc(&mut self, read_qc: ReadQc) {
102        self.read_qc += read_qc;
103    }
104
105    pub fn total_reads(&self) -> usize {
106        self.total_kept_reads() + self.total_dropped_reads()
107    }
108
109    pub fn total_kept_reads(&self) -> usize {
110        self.kept
111    }
112
113    pub fn total_dropped_reads(&self) -> usize {
114        self.dropped_by.values().sum()
115    }
116
117    /// Q30 fractions under their report names.
118    pub fn q30(&self) -> Q30Statistics {
119        Q30Statistics {
120            total: self.read_qc.q30_whole.rate(),
121            pid1: self.read_qc.q30_pid_a.rate(),
122            umi1: self.read_qc.q30_umi_a.rate(),
123            pid2: self.read_qc.q30_pid_b.rate(),
124            umi2: self.read_qc.q30_umi_b.rate(),
125        }
126    }
127
128    /// Drop tallies by reason name, most frequent first.
129    ///
130    /// Every drop reason appears, including the ones that never fired, so the
131    /// key set is the same on every run.
132    pub fn dropped_by_frequency(&self) -> Vec<(&'static str, usize)> {
133        let mut dropped_by_list: Vec<_> = DropReason::ALL
134            .iter()
135            .map(|reason| {
136                (
137                    reason.as_str(),
138                    self.dropped_by.get(reason).copied().unwrap_or_default(),
139                )
140            })
141            .collect();
142        dropped_by_list.sort_unstable_by_key(|&(reason, count)| (Reverse(count), reason));
143        dropped_by_list
144    }
145
146    /// Kept reads by `marker_1` id, most frequent first, ties by id.
147    ///
148    /// `panel` must be the one the run used, since the tallies are keyed by
149    /// position in it. Every panel marker appears, including those never seen,
150    /// so the key set is the panel on every run. The counts sum to
151    /// [`total_kept_reads`] and match the `marker_1` column of the Parquet.
152    ///
153    /// [`total_kept_reads`]: Self::total_kept_reads
154    pub fn marker1_by_frequency<'p>(&self, panel: &'p Panel) -> Vec<(&'p str, usize)> {
155        rank_markers(panel, &self.marker1_reads)
156    }
157
158    /// As [`marker1_by_frequency`](Self::marker1_by_frequency), for `marker_2`.
159    pub fn marker2_by_frequency<'p>(&self, panel: &'p Panel) -> Vec<(&'p str, usize)> {
160        rank_markers(panel, &self.marker2_reads)
161    }
162}

172    fn three_marker_panel() -> Panel {
173        let marker = |id: &str, s1: &str, s2: &str| PanelMarker {
174            marker_id: id.to_string(),
175            sequence_1: s1.to_string(),
176            sequence_2: s2.to_string(),

186    /// A `ReadQc` whose five regions carry distinct rates, so an accessor wired
187    /// to the wrong region reports a visibly wrong number rather than a
188    /// plausible one. Denominators are powers of two to keep every expected
189    /// rate exactly representable as an `f64`.
190    fn fingerprinted_qc() -> ReadQc {
191        let count = |q30_bases| Q30Count {
192            q30_bases,
193            total_bases: 8,
194        };
195        ReadQc {
196            q30_whole: count(1), // 0.125
197            q30_umi_a: count(2), // 0.25
198            q30_pid_a: count(3), // 0.375
199            q30_pid_b: count(4), // 0.5
200            q30_umi_b: count(5), // 0.625
201        }
202    }
203
204    /// A `ReadQc` carrying only a whole-read tally.
205    fn whole_read_qc(q30_bases: usize, total_bases: usize) -> ReadQc {
206        ReadQc {
207            q30_whole: Q30Count {
208                q30_bases,
209                total_bases,
210            },
211            ..Default::default()
212        }
213    }
