3    seq.contains(&b'N')
4}

6/// Returns whether the mean Phred+33 quality across the informative `regions`
7/// (pid + umi) is strictly below `min`.
8///
9/// Only the informative regions are considered; linker/uei bases are ignored.
10/// An empty total (e.g. missing quality) is treated as not-below (kept),
11/// matching the "missing quality keeps the read" policy when `qual()` is `None`.
12pub(super) fn avg_phred_below(regions: &[&[u8]], min: u8) -> bool {
13    let mut sum: u64 = 0;
14    let mut len: usize = 0;
15    for region in regions {
16        sum += region.iter().map(|&q| (q - 33) as u64).sum::<u64>();
17        len += region.len();
18    }
19    if len == 0 {
20        return false;
21    }
22    (sum as f64 / len as f64) < min as f64
23}

25/// Returns whether any single base accounts for strictly more than `max_freq`
26/// of the UMI length.
27pub(super) fn umi_base_freq_exceeds(umi: &[u8], max_freq: f64) -> bool {
28    if umi.is_empty() {
29        return false;
30    }
31    let mut counts = [0u32; 4];
32    for &b in umi {
33        let i = match b {
34            b'A' | b'a' => 0,
35            b'C' | b'c' => 1,
36            b'G' | b'g' => 2,
37            b'T' | b't' => 3,
38            _ => continue,
39        };
40        counts[i] += 1;
41    }
42    let max = counts.into_iter().max().unwrap_or(0);
43    (max as f64 / umi.len() as f64) > max_freq
44}

46/// Recommended maximum edit distance for the LBS–UMI overlap filter. Chosen
47/// against the observed contaminant UMIs (all caught at `k ≤ 6`, none at
48/// distance 7 or beyond).
49pub const DEFAULT_LBS_OVERLAP_MAX_EDITS: u8 = 6;

51/// Read-filtering and panel-matching configuration threaded through the
52/// edgelist pipeline.
53#[derive(Clone, Copy, Debug)]
54pub struct FilterConfig {
55    /// Maximum number of mismatches tolerated when matching a pid against the
56    /// panel. `None` means exact matching (threshold 0).
57    pub mismatch_threshold: Option<u8>,
58    /// Minimum average Phred quality across the informative regions; reads below
59    /// this drop as `LowQuality`. `None` disables the gate.
60    pub min_avg_quality: Option<u8>,
61    /// Maximum single-base frequency allowed in the UMI; reads above this drop
62    /// as `LowComplexity`. `None` disables the gate.
63    pub max_umi_base_freq: Option<f64>,
64    /// Maximum bounded infix Levenshtein distance for the LBS–UMI overlap
65    /// filter. A UMI is dropped when it aligns to any substring of an LBS
66    /// reference (or its reverse complement) within this many edits. `None`
67    /// disables the gate.
68    pub lbs_overlap_max_edits: Option<u8>,
69}
70
71impl Default for FilterConfig {
72    /// The LBS-overlap filter is on by default (like the neighbouring quality and
73    /// complexity gates in production), so the default enables it; the other
74    /// gates stay `None` (opt-in) as before.
75    fn default() -> Self {
76        Self {
77            mismatch_threshold: None,
78            min_avg_quality: None,
79            max_umi_base_freq: None,
80            lbs_overlap_max_edits: Some(DEFAULT_LBS_OVERLAP_MAX_EDITS),
81        }
82    }
83}

89    #[test]
90    fn has_n_detects_uppercase_n_anywhere() {
91        assert!(has_n(b"ACNT"));
92        assert!(has_n(b"NACGT"));
93        assert!(has_n(b"ACGTN"));
94    }
95
96    #[test]
97    fn has_n_is_false_without_uppercase_n() {
98        assert!(!has_n(b"ACGT"));
99        // Only the uppercase `N` sentinel counts; lowercase `n` is not treated
100        // as a missing base here.
101        assert!(!has_n(b"acnt"));
102    }
103
104    #[test]
105    fn has_n_empty_slice_is_false() {
106        assert!(!has_n(b""));
107    }
108
109    #[test]
110    fn avg_phred_below_true_when_mean_under_threshold() {
111        // '+' is Phred+33 for Q10; a mean of 10 is below 20.
112        assert!(avg_phred_below(&[b"++++"], 20));
113    }
114
115    #[test]
116    fn avg_phred_below_false_when_mean_at_or_above_threshold() {
117        // 'I' is Q40, well above 30.
118        assert!(!avg_phred_below(&[b"IIII"], 30));
119        // '5' is exactly Q20; the gate is strict (`<`), so Q20 is kept at min 20.
120        assert!(!avg_phred_below(&[b"5555"], 20));
121    }
122
123    #[test]
124    fn avg_phred_below_pools_all_regions_into_one_mean() {
125        // Two Q40 bases ('I') and two Q10 bases ('+') average to Q25.
126        assert!(avg_phred_below(&[b"II", b"++"], 30));
127        assert!(!avg_phred_below(&[b"II", b"++"], 20));
128    }
129
130    #[test]
131    fn avg_phred_below_empty_total_is_kept() {
132        // No regions, or only empty regions, means no quality signal: kept.
133        assert!(!avg_phred_below(&[], 40));
134        assert!(!avg_phred_below(&[b"", b""], 40));
135    }
136
137    #[test]
138    fn umi_base_freq_exceeds_true_for_low_complexity() {
139        // A homopolymer is 100% one base, above any threshold below 1.0.
140        assert!(umi_base_freq_exceeds(b"AAAA", 0.9));
141        // Lowercase bases are counted the same as uppercase.
142        assert!(umi_base_freq_exceeds(b"aaaa", 0.9));
143    }
144
145    #[test]
146    fn umi_base_freq_exceeds_false_for_diverse_umi() {
147        // Each base appears once: max frequency is 0.25.
148        assert!(!umi_base_freq_exceeds(b"ACGT", 0.9));
149    }
150
151    #[test]
152    fn umi_base_freq_exceeds_is_strict_at_boundary() {
153        // Three of four bases are 'A' -> frequency 0.75.
154        assert!(!umi_base_freq_exceeds(b"AAAC", 0.75));
155        assert!(umi_base_freq_exceeds(b"AAAC", 0.7));
156    }
157
158    #[test]
159    fn umi_base_freq_exceeds_counts_non_acgt_toward_length_only() {
160        // `N` bases are not tallied per-base but still count toward the length,
161        // so two of four positions are 'A' -> frequency 0.5.
162        assert!(!umi_base_freq_exceeds(b"AANN", 0.6));
163        assert!(umi_base_freq_exceeds(b"AANN", 0.4));
164    }
165
166    #[test]
167    fn umi_base_freq_exceeds_empty_umi_is_kept() {
168        assert!(!umi_base_freq_exceeds(b"", 0.5));
169    }
