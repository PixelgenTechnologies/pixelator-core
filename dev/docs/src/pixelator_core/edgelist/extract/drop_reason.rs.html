1/// Why a read pair was dropped during extraction.
2///
3/// Only the drop points that exist today are listed; future filters add
4/// variants.
5#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
6pub enum DropReason {
7    /// Either read contained an `N` base in the informative regions.
8    HasN,
9    /// The `pid-1` barcode was not found in the panel.
10    Pid1NotInPanel,
11    /// The `pid-2` barcode was not found in the panel.
12    Pid2NotInPanel,
13    /// The `pid-1` barcode matched two or more markers ambiguously.
14    Pid1Ambiguous,
15    /// The `pid-2` barcode matched two or more markers ambiguously.
16    Pid2Ambiguous,
17    /// The read is too short to contain all necessary regions
18    TooShort,
19    /// The read's average Phred quality is below the configured threshold in the informative regions.
20    LowQuality,
21    /// The UMI is low-complexity: one base accounts for more than the configured
22    /// fraction of its length.
23    LowComplexity,
24    /// The UMI aligns too closely to a linker binding site (LBS) sequence,
25    /// suggesting LBS contamination of the UMI region.
26    LbsOverlap,
27}
28
29impl DropReason {
30    /// Every variant, so a report can list the reasons that never fired too.
31    pub const ALL: [DropReason; 9] = [
32        DropReason::HasN,
33        DropReason::Pid1NotInPanel,
34        DropReason::Pid2NotInPanel,
35        DropReason::Pid1Ambiguous,
36        DropReason::Pid2Ambiguous,
37        DropReason::TooShort,
38        DropReason::LowQuality,
39        DropReason::LowComplexity,
40        DropReason::LbsOverlap,
41    ];
42
43    /// This reason's position in [`DropReason::ALL`], usable as a key into a
44    /// dictionary built from `ALL`'s names.
45    pub fn index(self) -> usize {
46        // Holds because `ALL` lists the variants in declaration order, which
47        // `index_is_the_position_in_all` checks.
48        self as usize
49    }
50
51    /// The reason's snake_case name
52    pub fn as_str(&self) -> &'static str {
53        // Renaming one changes the report schema, so treat these as fixed.
54        match self {
55            DropReason::HasN => "has_n",
56            DropReason::Pid1NotInPanel => "pid1_not_in_panel",
57            DropReason::Pid2NotInPanel => "pid2_not_in_panel",
58            DropReason::Pid1Ambiguous => "pid1_ambiguous",
59            DropReason::Pid2Ambiguous => "pid2_ambiguous",
60            DropReason::TooShort => "too_short",
61            DropReason::LowQuality => "low_quality",
62            DropReason::LowComplexity => "low_complexity",
63            DropReason::LbsOverlap => "lbs_overlap",
64        }
65    }
66}

74    #[test]
75    fn every_variant_reports_its_snake_case_name() {
76        for reason in DropReason::ALL {
77            let expected = match reason {
78                DropReason::HasN => "has_n",
79                DropReason::Pid1NotInPanel => "pid1_not_in_panel",
80                DropReason::Pid2NotInPanel => "pid2_not_in_panel",
81                DropReason::Pid1Ambiguous => "pid1_ambiguous",
82                DropReason::Pid2Ambiguous => "pid2_ambiguous",
83                DropReason::TooShort => "too_short",
84                DropReason::LowQuality => "low_quality",
85                DropReason::LowComplexity => "low_complexity",
86                DropReason::LbsOverlap => "lbs_overlap",
87            };
88            assert_eq!(reason.as_str(), expected);
89        }
90    }

92    #[test]
93    fn index_is_the_position_in_all() {
94        for (position, reason) in DropReason::ALL.into_iter().enumerate() {
95            assert_eq!(reason.index(), position, "{reason:?}");
96        }
97    }
98
99    #[test]
100    fn names_are_unique() {
101        let names: HashSet<&str> = DropReason::ALL.iter().map(|r| r.as_str()).collect();
102
103        assert_eq!(names.len(), DropReason::ALL.len());
104    }
105
106    /// Number of variants in the DropReason enum. Must be updated each time a
107    /// variant is added or removed from DropReason.
108    ///
109    /// In practice, this should always be the same as `DropReason::ALL.len()`,
110    /// as long as one remembers to update ALL when variants are added or removed.
111    ///
112    /// If one forgets, the test below will trigger either a compilation error or
113    /// fail.
114    const N_VARIANTS: usize = 9;
115
116    /// The column claimed by a single variant.
117    const fn column(index: usize) -> [usize; N_VARIANTS] {
118        let mut claimed = [0; N_VARIANTS];
119        claimed[index] = 1;
120        claimed
121    }
122
123    #[test]
124    fn all_lists_every_variant_exactly_once() {
125        // Each arm must claim a column of its own, and each index must stay
126        // inside a `const` block: that is what turns a new variant into a
127        // compile error until `N_VARIANTS` grows, and a `N_VARIANTS` that outgrows
128        // `ALL` into the failure below. Calling `column` outside `const` moves
129        // the bounds check to runtime and might never happen if the variant is
130        // missing from `ALL`.
131        let mut tally = [0usize; N_VARIANTS];
132        for reason in DropReason::ALL {
133            let claimed = match reason {
134                DropReason::HasN => const { column(0) },
135                DropReason::Pid1NotInPanel => const { column(1) },
136                DropReason::Pid2NotInPanel => const { column(2) },
137                DropReason::Pid1Ambiguous => const { column(3) },
138                DropReason::Pid2Ambiguous => const { column(4) },
139                DropReason::TooShort => const { column(5) },
140                DropReason::LowQuality => const { column(6) },
141                DropReason::LowComplexity => const { column(7) },
142                DropReason::LbsOverlap => const { column(8) },
143            };
144            for (total, hit) in tally.iter_mut().zip(claimed) {
145                *total += hit;
146            }
147        }
148
149        assert!(
150            tally.iter().all(|&count| count == 1),
151            "every variant must appear in DropReason::ALL exactly once, got {tally:?}"
152        );
153    }
