1//! How a raw read is laid out: the vocabulary for locating a region within it.
2//!
3//! A region's position depends on two independent axes - which conjugate it
4//! belongs to ([`ConjugateSide`]) and which orientation the read was sequenced
5//! in ([`Strand`]) - and [`RegionLayout`] resolves that pair into byte offsets.
6//! [`Read`] is the unit those offsets apply to.
7//!
8//! These types are shared vocabulary rather than the property of any one stage:
9//! extraction and quality profiling both address reads this way, so neither owns
10//! them. Nothing here depends on the panel or on extraction, which keeps the
11//! dependency arrows pointing inward from both consumers.
12
13mod conjugate;
14mod layout;
15mod read;
16mod strand;
17
18pub use conjugate::ConjugateSide;
19pub use read::Read;
20pub use strand::Strand;
21
22// Narrower than its neighbours: the resolved offsets are an internal detail
23// shared between pipeline stages, not part of the crate's surface.
24pub(crate) use layout::RegionLayout;
