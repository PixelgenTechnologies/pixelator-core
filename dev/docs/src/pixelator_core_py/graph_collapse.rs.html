20/// One edgelist row: `(component, umi1, umi2)`.
21type EdgelistRow = (u32, u64, u64);
22
23/// One collapsed UMI: `(component, original, corrected)`.
24type Rename = (u32, u64, u64);
25
26/// Collapse likely sequencing-error UMIs in an edgelist.
27///
28/// The three arrays are the columns of the edgelist, one entry per edge. All
29/// edges are held in memory, so the caller bounds memory by passing one batch
30/// of whole components at a time. Rows may come in any order: they are grouped
31/// by `component` and sorted before each graph is built, so the result does not
32/// depend on the order. Components are searched in parallel.
33///
34/// Args:
35///     umi1: 1-D uint64 numpy array of 2-bit-encoded `umi1` values.
36///     umi2: 1-D uint64 numpy array of 2-bit-encoded `umi2` values.
37///     component: 1-D uint32 numpy array with the component of each edge. It
38///         only has to tell components apart; the caller maps it back to a name.
39///     n_threads: Number of worker threads for the search.
40///     bfs_threshold: Maximum graph distance searched for a correction target.
41///     hamming_threshold: Maximum Hamming distance between an investigated UMI
42///         and its correction target.
43///     max_degree: Nodes with degree above this value are not searched. The
44///         default, 1, investigates only leaves.
45///
46/// Returns:
47///     A `(renames, stats)` tuple.
48///
49///     `renames` is a dict of three equally long 1-D numpy arrays, one entry per
50///     collapsed UMI, sorted by `component` and then `original`:
51///
52///     - `component` (uint32): the component the UMI was collapsed in.
53///     - `original` (uint64): the collapsed UMI.
54///     - `corrected` (uint64): the UMI it is collapsed into.
55///
56///     `stats` is a dict of what the search counted, over all components:
57///
58///     - `total_components`: components processed.
59///     - `nodes_considered`: nodes with degree at most `max_degree`.
60///     - `targets_found`: kept matches. A match dropped because it would close
61///       a loop is not counted, and neither is a pair that the caller later
62///       discards (for example for differing markers).
63///     - `bfs_distance_distribution`: `{hops: count}` over kept matches.
64///     - `hamming_distance_distribution`: `{distance: count}` over kept matches.
65///
66///     Both distributions list only distances that occurred, in ascending
67///     order. Each distance is that of the match as found: for a chain
68///     `A -> B -> H` it is the distance from A to B and from B to H, not from
69///     A to H.
70///
71/// Raises:
72///     ValueError: If the arrays differ in length.
73#[pyfunction(
74    name = "graph_collapse",
75    signature = (
76        umi1,
77        umi2,
78        component,
79        n_threads,
80        bfs_threshold=6,
81        hamming_threshold=2,
82        max_degree=1,
83    )
84)]
85#[allow(clippy::too_many_arguments)]
86pub fn py_graph_collapse<'py>(
87    py: Python<'py>,
88    umi1: PyReadonlyArray1<'py, u64>,
89    umi2: PyReadonlyArray1<'py, u64>,
90    component: PyReadonlyArray1<'py, u32>,
91    n_threads: usize,
92    bfs_threshold: u64,
93    hamming_threshold: u8,
94    max_degree: u64,
95) -> PyResult<(Bound<'py, PyDict>, Bound<'py, PyDict>)> {
96    let (umi1, umi2, component) = (umi1.as_array(), umi2.as_array(), component.as_array());
97    if umi1.len() != umi2.len() || umi1.len() != component.len() {
98        return Err(PyValueError::new_err(format!(
99            "umi1, umi2 and component must have the same length, got {}, {} and {}",
100            umi1.len(),
101            umi2.len(),
102            component.len()
103        )));
104    }
105    // Copied while the GIL is held, so Python cannot change the arrays while
106    // the search runs without it.
107    let rows: Vec<EdgelistRow> = izip!(component.iter(), umi1.iter(), umi2.iter())
108        .map(|(&component, &umi1, &umi2)| (component, umi1, umi2))
109        .collect();
110
111    // `graph_collapse` parallelizes on the current Rayon pool. A local pool
112    // keeps that at `n_threads` instead of the CPU count.
113    let pool = ThreadPoolBuilder::new()
114        .num_threads(n_threads)
115        .build()
116        .map_err(to_py_err)?;
117    let (renames, stats) = py.detach(move || {
118        pool.install(|| collapse_edgelist(rows, bfs_threshold, hamming_threshold, max_degree))
119    });
120
121    Ok((renames_to_dict(py, renames)?, stats_to_dict(py, &stats)?))
122}

124/// Run graph collapse on every component of the edgelist.
125///
126/// Each component is loaded into its own graph and collapsed independently,
127/// in parallel.
128///
129/// Returns the collapsed UMIs, sorted by component and then original UMI,
130/// together with the stats summed over components.
131fn collapse_edgelist(
132    mut rows: Vec<EdgelistRow>,
133    bfs_threshold: u64,
134    hamming_threshold: u8,
135    max_degree: u64,
136) -> (Vec<Rename>, CollapseStats) {
137    // Node indices and neighbor order follow the edge order, and they decide
138    // which match the search finds first. Sorting groups each component into
139    // one run and makes the result independent of the row order.
140    rows.par_sort_unstable();
141
142    let per_component: Vec<(Vec<Rename>, CollapseStats)> = rows
143        .par_chunk_by(|a, b| a.0 == b.0)
144        .map(|component_rows| {
145            collapse_component(component_rows, bfs_threshold, hamming_threshold, max_degree)
146        })
147        .collect();
148
149    let mut stats = CollapseStats::default();
150    let mut renames = Vec::new();
151    for (component_renames, component_stats) in per_component {
152        stats.add_components(1);
153        stats += component_stats;
154        renames.extend(component_renames);
155    }
156    (renames, stats)
157}

159/// Collapse one component, given its rows sorted by `(umi1, umi2)`.
160///
161/// Returns the component's renames, sorted by original UMI, and its stats.
162fn collapse_component(
163    rows: &[EdgelistRow],
164    bfs_threshold: u64,
165    hamming_threshold: u8,
166    max_degree: u64,
167) -> (Vec<Rename>, CollapseStats) {
168    let component = rows[0].0;
169    let umi_pairs = rows
170        .iter()
171        .map(|&(_, umi1, umi2)| (umi1 as UMI, umi2 as UMI));
172    let umi_mapping = UmiToNodeIndexMapping::from_umi_pairs(umi_pairs.clone());
173    let num_nodes = umi_mapping.get_num_of_nodes();
174    let edges = umi_mapping.map_umi_pair_iterator_to_edge(umi_pairs);
175    let graph = Graph::<u8>::from_edges(edges, num_nodes);
176
177    let (targets, stats) = graph_collapse(
178        &graph,
179        &umi_mapping,
180        bfs_threshold as usize,
181        hamming_threshold,
182        max_degree as usize,
183    );
184    let mut renames: Vec<Rename> = targets
185        .into_iter()
186        .map(|(original, corrected)| (component, original as u64, corrected as u64))
187        .collect();
188    renames.sort_unstable();
189    (renames, stats)
190}

192/// The renames as a dict of `component`, `original` and `corrected` numpy arrays.
193fn renames_to_dict<'py>(py: Python<'py>, renames: Vec<Rename>) -> PyResult<Bound<'py, PyDict>> {
194    let mut component = Vec::with_capacity(renames.len());
195    let mut original = Vec::with_capacity(renames.len());
196    let mut corrected = Vec::with_capacity(renames.len());
197    for (c, o, r) in renames {
198        component.push(c);
199        original.push(o);
200        corrected.push(r);
201    }
202    let out = PyDict::new(py);
203    out.set_item("component", component.into_pyarray(py))?;
204    out.set_item("original", original.into_pyarray(py))?;
205    out.set_item("corrected", corrected.into_pyarray(py))?;
206    Ok(out)
207}

209/// Mirrors [`CollapseStats`] as a plain dict, with the keys listed in the
210/// docstring of [`py_graph_collapse`].
211fn stats_to_dict<'py>(py: Python<'py>, stats: &CollapseStats) -> PyResult<Bound<'py, PyDict>> {
212    let out = PyDict::new(py);
213    out.set_item("total_components", stats.total_components)?;
214    out.set_item("nodes_considered", stats.nodes_considered)?;
215    out.set_item("targets_found", stats.targets_found)?;
216
217    let bfs_distance_dict = PyDict::new(py);
218    for (d, c) in stats.bfs_distance_counts() {
219        bfs_distance_dict.set_item(d, c)?;
220    }
221    out.set_item("bfs_distance_distribution", bfs_distance_dict)?;
222
223    let hamming_distance_dict = PyDict::new(py);
224    for (d, c) in stats.hamming_distance_counts() {
225        hamming_distance_dict.set_item(d, c)?;
226    }
227    out.set_item("hamming_distance_distribution", hamming_distance_dict)?;
228
229    Ok(out)
230}

242    #[test]
243    fn chain_collapses_within_its_component_only() {
244        // A chain A -> B -> H in component 7. Its rows are interleaved with
245        // component 3, so grouping cannot rely on the row order. CCAA is one
246        // symbol from CAAA, and CAAA is one symbol from both CCAA and AAAA.
247        // Rows are sorted before the search, so CAAA meets AAAA, the chain's
248        // end, first. Component 3 is one edge between two distant UMIs, which
249        // has no collapse target. M and N are umi1 and every other UMI is
250        // umi2, so none sits on both sides.
251        let umi_m = umi("GGGG");
252        let umi_h = umi("AAAA");
253        let umi_a = umi("CCAA");
254        let umi_b = umi("CAAA");
255        let umi_n = umi("TTTT");
256        let rows = vec![
257            (7, umi_m, umi_h),
258            (3, umi("ACAC"), umi("GTGT")),
259            (7, umi_m, umi_b),
260            (7, umi_n, umi_h),
261            (7, umi_m, umi_a),
262        ];
263
264        let (renames, stats) = collapse_edgelist(rows, 2, 1, 1);
265
266        let mut expected = vec![(7, umi_a, umi_h), (7, umi_b, umi_h)];
267        expected.sort_unstable();
268        assert_eq!(renames, expected);
269        // Component 7 has three leaves (A, B and N) and component 3 has two.
270        // Only the chain in component 7 finds matches.
271        assert_eq!(stats.total_components, 2);
272        assert_eq!(stats.nodes_considered, 5);
273        assert_eq!(stats.targets_found, 2);
274        assert_eq!(stats.bfs_distance_counts(), [(2, 2)]);
275        assert_eq!(stats.hamming_distance_counts(), [(1, 2)]);
276    }

279    fn same_umis_in_two_components_collapse_separately() {
280        // The same leaf chain appears in components 1 and 2. Each component is
281        // its own graph, so each reports its own renames.
282        let umi_m = umi("GGGG");
283        let umi_h = umi("AAAA");
284        let umi_a = umi("CAAA");
285        let umi_n = umi("TTTT");
286        let rows = vec![
287            (2, umi_m, umi_h),
288            (1, umi_m, umi_h),
289            (2, umi_m, umi_a),
290            (1, umi_m, umi_a),
291            (2, umi_n, umi_h),
292            (1, umi_n, umi_h),
293        ];
294
295        let (renames, stats) = collapse_edgelist(rows, 2, 1, 1);

297        assert_eq!(renames, vec![(1, umi_a, umi_h), (2, umi_a, umi_h)]);

299        assert_eq!(stats.targets_found, 2);
300    }
