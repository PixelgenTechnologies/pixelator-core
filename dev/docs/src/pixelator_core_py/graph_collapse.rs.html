23/// One edgelist row: `(component, umi1, umi2)`.
24type EdgelistRow = (u32, u64, u64);
25
26/// One collapsed UMI: `(component, original, corrected)`.
27type Rename = (u32, u64, u64);
28
29/// Collapse likely sequencing-error UMIs in an edgelist.
30///
31/// The rows are read into memory, so the caller bounds memory by passing one
32/// batch of components at a time. Rows may come in any order, and a component
33/// may span several record batches: rows are grouped by `component` and sorted
34/// before each graph is built, so the result does not depend on the order.
35/// Components are searched in parallel.
36///
37/// Args:
38///     edgelist_reader: A pyarrow `RecordBatchReader` with `umi1` and `umi2`
39///         (uint64) and `component` (uint32) columns. `component` only has to
40///         tell components apart; the caller maps it back to a name.
41///     n_threads: Number of worker threads for the search.
42///     bfs_threshold: Maximum graph distance searched for a correction target.
43///     hamming_threshold: Maximum Hamming distance between an investigated UMI
44///         and its correction target.
45///     max_degree: Nodes with degree above this value are not searched. The
46///         default, 1, investigates only leaves.
47///
48/// Returns:
49///     A `(table, stats)` tuple.
50///
51///     `table` is a pyarrow `Table` with a `component` uint32 column and
52///     `original` and `corrected` uint64 columns, one row per collapsed UMI,
53///     sorted by `component` and then `original`.
54///
55///     `stats` is a dict of what the search counted, over all components:
56///
57///     - `total_components`: components processed.
58///     - `nodes_considered`: nodes with degree at most `max_degree`.
59///     - `targets_found`: kept matches. A match dropped because it would close
60///       a loop is not counted, and neither is a pair that the caller later
61///       discards (for example for differing markers).
62///     - `bfs_distance_distribution`: `{hops: count}` over kept matches.
63///     - `hamming_distance_distribution`: `{distance: count}` over kept matches.
64///
65///     Both distributions list only distances that occurred, in ascending
66///     order. Each distance is that of the match as found: for a chain
67///     `A -> B -> H` it is the distance from A to B and from B to H, not from
68///     A to H.
69#[pyfunction(
70    name = "graph_collapse",
71    signature = (
72        edgelist_reader,
73        n_threads,
74        bfs_threshold=6,
75        hamming_threshold=2,
76        max_degree=1,
77    )
78)]
79pub fn py_graph_collapse<'py>(
80    py: Python<'py>,
81    edgelist_reader: PyArrowType<arrow::ffi_stream::ArrowArrayStreamReader>,
82    n_threads: usize,
83    bfs_threshold: u64,
84    hamming_threshold: u8,
85    max_degree: u64,
86) -> PyResult<(PyArrowType<Table>, Bound<'py, PyDict>)> {
87    // `graph_collapse` parallelizes on the current Rayon pool. A local pool
88    // keeps that at `n_threads` instead of the CPU count. The GIL is released
89    // for the duration: the pool thread reads the Arrow stream, whose
90    // callbacks take the GIL themselves, and would deadlock if we still held it.
91    let pool = ThreadPoolBuilder::new()
92        .num_threads(n_threads)
93        .build()
94        .map_err(to_py_err)?;
95    let PyArrowType(edgelist_reader) = edgelist_reader;
96    let (table, stats) = py
97        .detach(move || {
98            pool.install(|| {
99                run_graph_collapse(
100                    edgelist_reader,
101                    bfs_threshold,
102                    hamming_threshold,
103                    max_degree,
104                )
105            })
106        })
107        .map_err(arrow_to_py_err)?;
108
109    Ok((PyArrowType(table), stats_to_dict(py, &stats)?))
110}

112/// An input that does not match the expected columns is a `ValueError`; any
113/// other failure, such as one raised while reading the stream, a `RuntimeError`.
114fn arrow_to_py_err(err: ArrowError) -> PyErr {
115    match err {
116        ArrowError::SchemaError(message) => PyValueError::new_err(message),
117        other => to_py_err(other),
118    }
119}

121/// Run graph collapse on every component of the edgelist.
122///
123/// Each component is loaded into its own graph and collapsed independently,
124/// in parallel. Reading stops at the first batch that fails, and that error is
125/// returned.
126///
127/// Returns the collapsed UMIs together with the stats summed over components.
128fn run_graph_collapse(
129    edgelist_reader: impl Iterator<Item = Result<RecordBatch, ArrowError>>,
130    bfs_threshold: u64,
131    hamming_threshold: u8,
132    max_degree: u64,
133) -> Result<(Table, CollapseStats), ArrowError> {
134    let mut rows = read_rows(edgelist_reader)?;
135    // Node indices and neighbor order follow the edge order, and they decide
136    // which match the search finds first. Sorting groups each component into
137    // one run and makes the result independent of the row order.
138    rows.par_sort_unstable();
139
140    let per_component: Vec<(Vec<Rename>, CollapseStats)> = rows
141        .chunk_by(|a, b| a.0 == b.0)
142        .collect::<Vec<_>>()
143        .into_par_iter()
144        .map(|component_rows| {
145            collapse_component(component_rows, bfs_threshold, hamming_threshold, max_degree)
146        })
147        .collect();
148
149    let mut stats = CollapseStats::default();
150    let mut renames = Vec::new();
151    for (component_renames, component_stats) in per_component {
152        stats.add_components(1);
153        stats += component_stats;
154        renames.extend(component_renames);
155    }
156    arrow_table(renames).map(|table| (table, stats))
157}

159/// Read every row of the stream into memory.
160///
161/// Fails with a schema error if a column is missing, has the wrong type, or
162/// contains a null.
163fn read_rows(
164    edgelist_reader: impl Iterator<Item = Result<RecordBatch, ArrowError>>,
165) -> Result<Vec<EdgelistRow>, ArrowError> {
166    let mut rows = Vec::new();
167    for batch in edgelist_reader {
168        let batch = batch?;
169        let component = uint32_column(&batch, "component")?;
170        let umi1 = uint64_column(&batch, "umi1")?;
171        let umi2 = uint64_column(&batch, "umi2")?;
172        rows.extend(
173            (0..batch.num_rows())
174                .map(|row| (component.value(row), umi1.value(row), umi2.value(row))),
175        );
176    }
177    Ok(rows)
178}

180/// Collapse one component, given its rows sorted by `(umi1, umi2)`.
181///
182/// Returns the component's renames, sorted by original UMI, and its stats.
183fn collapse_component(
184    rows: &[EdgelistRow],
185    bfs_threshold: u64,
186    hamming_threshold: u8,
187    max_degree: u64,
188) -> (Vec<Rename>, CollapseStats) {
189    let component = rows[0].0;
190    let umi_pairs = rows
191        .iter()
192        .map(|&(_, umi1, umi2)| (umi1 as UMI, umi2 as UMI));
193    let umi_mapping = UmiToNodeIndexMapping::from_umi_pairs(umi_pairs.clone());
194    let num_nodes = umi_mapping.get_num_of_nodes();
195    let edges = umi_mapping.map_umi_pair_iterator_to_edge(umi_pairs);
196    let graph = Graph::<u8>::from_edges(edges, num_nodes);
197
198    let (targets, stats) = graph_collapse(
199        &graph,
200        &umi_mapping,
201        bfs_threshold as usize,
202        hamming_threshold,
203        max_degree as usize,
204    );
205    let mut renames: Vec<Rename> = targets
206        .into_iter()
207        .map(|(original, corrected)| (component, original as u64, corrected as u64))
208        .collect();
209    renames.sort_unstable();
210    (renames, stats)
211}

233/// Get a column by name, checking its type and that it has no nulls.
234fn non_null_column<'a>(
235    batch: &'a RecordBatch,
236    name: &str,
237    data_type: &DataType,
238) -> Result<&'a dyn Array, ArrowError> {
239    let column = batch
240        .column_by_name(name)
241        .ok_or_else(|| ArrowError::SchemaError(format!("Could not find {name} column in data")))?;
242    if column.data_type() != data_type {
243        return Err(ArrowError::SchemaError(format!(
244            "Column {name} must be {data_type}, got {}",
245            column.data_type()
246        )));
247    }
248    if column.null_count() > 0 {
249        return Err(ArrowError::SchemaError(format!(
250            "Column {name} contains {} null values",
251            column.null_count()
252        )));
253    }
254    Ok(column.as_ref())
255}

257/// Build a single-batch table with non-null `component` (UInt32) and
258/// `original` and `corrected` (UInt64) columns.
259fn arrow_table(renames: Vec<Rename>) -> Result<Table, ArrowError> {
260    let schema = Arc::new(Schema::new(vec![
261        Field::new("component", DataType::UInt32, false),
262        Field::new("original", DataType::UInt64, false),
263        Field::new("corrected", DataType::UInt64, false),
264    ]));
265    let (component, (original, corrected)): (Vec<u32>, (Vec<u64>, Vec<u64>)) = renames
266        .into_iter()
267        .map(|(component, original, corrected)| (component, (original, corrected)))
268        .unzip();
269    let batch = RecordBatch::try_new(
270        schema.clone(),
271        vec![
272            Arc::new(UInt32Array::from(component)),
273            Arc::new(UInt64Array::from(original)),
274            Arc::new(UInt64Array::from(corrected)),
275        ],
276    )?;
277    Table::try_new(vec![batch], schema)
278}

280/// Mirrors [`CollapseStats`] as a plain dict, with the keys listed in the
281/// docstring of [`py_graph_collapse`].
282fn stats_to_dict<'py>(py: Python<'py>, stats: &CollapseStats) -> PyResult<Bound<'py, PyDict>> {
283    let out = PyDict::new(py);
284    out.set_item("total_components", stats.total_components)?;
285    out.set_item("nodes_considered", stats.nodes_considered)?;
286    out.set_item("targets_found", stats.targets_found)?;
287
288    let bfs_distance_dict = PyDict::new(py);
289    for (d, c) in stats.bfs_distance_counts() {
290        bfs_distance_dict.set_item(d, c)?;
291    }
292    out.set_item("bfs_distance_distribution", bfs_distance_dict)?;
293
294    let hamming_distance_dict = PyDict::new(py);
295    for (d, c) in stats.hamming_distance_counts() {
296        hamming_distance_dict.set_item(d, c)?;
297    }
298    out.set_item("hamming_distance_distribution", hamming_distance_dict)?;
299
300    Ok(out)
301}

338    }

340    #[test]
341    fn chain_collapses_within_its_component_only() {
342        // A chain A -> B -> H in component 7. Its rows are split across
343        // batches and interleaved with component 3, so grouping cannot rely on
344        // the row order. CCAA is one symbol from CAAA, and CAAA is one symbol
345        // from both CCAA and AAAA. Rows are sorted before the search, so CAAA
346        // meets AAAA, the chain's end, first. Component 3 is one edge between
347        // two distant UMIs, which has no collapse target. M and N are umi1 and
348        // every other UMI is umi2, so none sits on both sides.
349        let umi_m = umi("GGGG");
350        let umi_h = umi("AAAA");
351        let umi_a = umi("CCAA");
352        let umi_b = umi("CAAA");
353        let umi_n = umi("TTTT");
354        let batches = [
355            batch(&[7, 3], &[umi_m, umi("ACAC")], &[umi_h, umi("GTGT")]),
356            batch(&[7, 7, 7], &[umi_m, umi_n, umi_m], &[umi_b, umi_h, umi_a]),
357        ];
358
359        let (table, stats) = run_graph_collapse(batches.into_iter().map(Ok), 2, 1, 1).unwrap();
360
361        let mut expected = vec![(7, umi_a, umi_h), (7, umi_b, umi_h)];
362        expected.sort_unstable();
363        assert_eq!(renames(&table), expected);
364        // Component 7 has three leaves (A, B and N) and component 3 has two.
365        // Only the chain in component 7 finds matches.
366        assert_eq!(stats.total_components, 2);
367        assert_eq!(stats.nodes_considered, 5);
368        assert_eq!(stats.targets_found, 2);
369        assert_eq!(stats.bfs_distance_counts(), [(2, 2)]);
370        assert_eq!(stats.hamming_distance_counts(), [(1, 2)]);
371    }
