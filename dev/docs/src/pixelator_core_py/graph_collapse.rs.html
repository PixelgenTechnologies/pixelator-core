26/// One edgelist row: `(umi1, umi2, component)`.
27type EdgelistRow = (u64, u64, String);
28
29/// Collapse likely sequencing-error UMIs in an edgelist.
30///
31/// The edgelist must be sorted by `component`, so each component is a
32/// contiguous run of rows. A component may span several record batches. The
33/// rows within a component may come in any order: they are sorted before the
34/// graph is built, so the result does not depend on it.
35///
36/// Args:
37///     edgelist_reader: A pyarrow `RecordBatchReader` with `umi1` and `umi2`
38///         (uint64) and `component` (string) columns, sorted by `component`.
39///     n_threads: Number of worker threads for the per-component search.
40///     bfs_threshold: Maximum graph distance searched for a correction target.
41///     hamming_threshold: Maximum Hamming distance between an investigated UMI
42///         and its correction target.
43///     max_degree: Nodes with degree above this value are not searched. The
44///         default, 1, investigates only leaves.
45///
46/// Returns:
47///     A `(table, stats)` tuple.
48///
49///     `table` is a pyarrow `Table` with `original` and `corrected` uint64
50///     columns, one row per collapsed UMI.
51///
52///     `stats` is a dict of what the search counted, over all components:
53///
54///     - `total_components`: components processed.
55///     - `nodes_considered`: nodes with degree at most `max_degree`.
56///     - `targets_found`: kept matches. A match dropped because it would close
57///       a loop is not counted, and neither is a pair that the caller later
58///       discards (for example for differing markers).
59///     - `bfs_distance_distribution`: `{hops: count}` over kept matches.
60///     - `hamming_distance_distribution`: `{distance: count}` over kept matches.
61///
62///     Both distributions list only distances that occurred, in ascending
63///     order. Each distance is that of the match as found: for a chain
64///     `A -> B -> H` it is the distance from A to B and from B to H, not from
65///     A to H.
66#[pyfunction(
67    name = "graph_collapse",
68    signature = (
69        edgelist_reader,
70        n_threads,
71        bfs_threshold=6,
72        hamming_threshold=2,
73        max_degree=1,
74    )
75)]
76pub fn py_graph_collapse<'py>(
77    py: Python<'py>,
78    edgelist_reader: PyArrowType<arrow::ffi_stream::ArrowArrayStreamReader>,
79    n_threads: usize,
80    bfs_threshold: u64,
81    hamming_threshold: u8,
82    max_degree: u64,
83) -> PyResult<(PyArrowType<Table>, Bound<'py, PyDict>)> {
84    // `graph_collapse` parallelizes on the current Rayon pool. A local pool
85    // keeps that at `n_threads` instead of the CPU count. The GIL is released
86    // for the duration: the pool thread reads the Arrow stream, whose
87    // callbacks take the GIL themselves, and would deadlock if we still held it.
88    let pool = ThreadPoolBuilder::new()
89        .num_threads(n_threads)
90        .build()
91        .map_err(to_py_err)?;
92    let PyArrowType(edgelist_reader) = edgelist_reader;
93    let (table, stats) = py
94        .detach(move || {
95            pool.install(|| {
96                run_graph_collapse(
97                    edgelist_reader,
98                    bfs_threshold,
99                    hamming_threshold,
100                    max_degree,
101                )
102            })
103        })
104        .map_err(to_py_err)?;
105
106    Ok((PyArrowType(table), stats_to_dict(py, &stats)?))
107}

109/// Run graph collapse on every component of the edgelist.
110///
111/// Each component is loaded into its own graph and collapsed independently.
112/// Reading stops at the first batch that fails, and that error is returned.
113///
114/// Returns the collapsed pairs together with the stats summed over components.
115fn run_graph_collapse(
116    edgelist_reader: impl Iterator<Item = Result<RecordBatch, ArrowError>>,
117    bfs_threshold: u64,
118    hamming_threshold: u8,
119    max_degree: u64,
120) -> Result<(Table, CollapseStats), ArrowError> {
121    let mut global_stats = CollapseStats::default();
122    let (original, corrected): (Vec<u64>, Vec<u64>) =
123        edgelist_reader.process_results(|batches| {
124            batches_to_graph_components(batches)
125                .into_iter()
126                .flat_map(|(_, component_edges)| {
127                    let mut umi_pairs: Vec<_> = component_edges
128                        .map(|(umi1, umi2, _)| (umi1 as UMI, umi2 as UMI))
129                        .collect();
130                    // Node indices and neighbor order follow the edge order, and they
131                    // decide which match the search finds first. Sorting makes the result
132                    // independent of the row order within a component.
133                    umi_pairs.par_sort_unstable();
134
135                    let umi_mapping =
136                        UmiToNodeIndexMapping::from_umi_pairs(umi_pairs.iter().copied());
137                    let num_nodes = umi_mapping.get_num_of_nodes();
138                    let edges = umi_mapping.map_umi_pair_iterator_to_edge(umi_pairs);
139                    let graph = Graph::<u8>::from_edges(edges, num_nodes);
140
141                    let (targets, local_stats) = graph_collapse(
142                        &graph,
143                        &umi_mapping,
144                        bfs_threshold as usize,
145                        hamming_threshold,
146                        max_degree as usize,
147                    );
148                    global_stats.add_components(1);
149                    global_stats += local_stats;
150                    targets
151                })
152                .map(|(original, corrected)| (original as u64, corrected as u64))
153                .unzip()
154        })?;
155
156    arrow_table(original, corrected).map(|table| (table, global_stats))
157}

159/// Flatten the batches into `(umi1, umi2, component)` rows grouped by component.
160///
161/// Consecutive rows with the same `component` form one group, regardless of
162/// batch boundaries. The groups must be consumed in order, so the result is
163/// iterated with `(&result).into_iter()` or `for (component, rows) in &result`.
164///
165/// Panics if a column is missing, has the wrong type, or contains a null.
166fn batches_to_graph_components(
167    batch_iter: impl Iterator<Item = RecordBatch>,
168) -> ChunkBy<String, impl Iterator<Item = EdgelistRow>, impl FnMut(&EdgelistRow) -> String> {
169    batch_iter
170        .flat_map(|batch| {
171            let umi1 = uint64_column(&batch, "umi1");
172            let umi2 = uint64_column(&batch, "umi2");
173            let component = utf8_column(&batch, "component");
174            let rows = umi1.len();
175            (0..rows).map(move |row| {
176                (
177                    umi1.value(row),
178                    umi2.value(row),
179                    component.value(row).to_string(),
180                )
181            })
182        })
183        .chunk_by(|(_, _, component)| component.clone())
184}

186/// Get a non-null UInt64 column by name.
187///
188/// Panics if the column is missing, is not UInt64, or contains a null.
189fn uint64_column(batch: &RecordBatch, name: &str) -> PrimitiveArray<UInt64Type> {
190    let column = batch
191        .column_by_name(name)
192        .unwrap_or_else(|| panic!("Could not find {name} column in data"))
193        .as_primitive_opt::<UInt64Type>()
194        .unwrap_or_else(|| panic!("Column {name} is not UInt64"));
195    assert_eq!(column.null_count(), 0, "{name} contains a null");
196    column.clone()
197}

199/// Get a non-null Utf8 column by name.
200///
201/// Panics if the column is missing, is not Utf8, or contains a null.
202fn utf8_column(batch: &RecordBatch, name: &str) -> StringArray {
203    let column = batch
204        .column_by_name(name)
205        .unwrap_or_else(|| panic!("Could not find {name} column in data"))
206        .as_string_opt::<i32>()
207        .unwrap_or_else(|| panic!("Column {name} is not Utf8"));
208    assert_eq!(column.null_count(), 0, "{name} contains a null");
209    column.clone()
210}

212/// Build a single-batch table with non-null `original` and `corrected` UInt64 columns.
213fn arrow_table(original: Vec<u64>, corrected: Vec<u64>) -> Result<Table, ArrowError> {
214    let schema = Arc::new(Schema::new(vec![
215        Field::new("original", DataType::UInt64, false),
216        Field::new("corrected", DataType::UInt64, false),
217    ]));
218    let batch = RecordBatch::try_new(
219        schema.clone(),
220        vec![
221            Arc::new(UInt64Array::from(original)),
222            Arc::new(UInt64Array::from(corrected)),
223        ],
224    )?;
225    Table::try_new(vec![batch], schema)
226}

228/// Mirrors [`CollapseStats`] as a plain dict, with the keys listed in the
229/// docstring of [`py_graph_collapse`].
230fn stats_to_dict<'py>(py: Python<'py>, stats: &CollapseStats) -> PyResult<Bound<'py, PyDict>> {
231    let out = PyDict::new(py);
232    out.set_item("total_components", stats.total_components)?;
233    out.set_item("nodes_considered", stats.nodes_considered)?;
234    out.set_item("targets_found", stats.targets_found)?;
235
236    let bfs_distance_dict = PyDict::new(py);
237    for (d, c) in stats.bfs_distance_counts() {
238        bfs_distance_dict.set_item(d, c)?;
239    }
240    out.set_item("bfs_distance_distribution", bfs_distance_dict)?;
241
242    let hamming_distance_dict = PyDict::new(py);
243    for (d, c) in stats.hamming_distance_counts() {
244        hamming_distance_dict.set_item(d, c)?;
245    }
246    out.set_item("hamming_distance_distribution", hamming_distance_dict)?;
247
248    Ok(out)
249}

279    fn component_groups(batches: impl Iterator<Item = RecordBatch>) -> Vec<Vec<(u64, u64)>> {
280        let grouped = batches_to_graph_components(batches);
281        (&grouped)
282            .into_iter()
283            .map(|(_, rows)| rows.map(|(umi1, umi2, _)| (umi1, umi2)).collect())
284            .collect()
285    }

287    fn collapsed_pairs(table: &Table) -> HashSet<(u64, u64)> {
288        let batch = &table.record_batches()[0];
289        let original = batch.column(0).as_primitive::<UInt64Type>().values();
290        let corrected = batch.column(1).as_primitive::<UInt64Type>().values();
291        original
292            .iter()
293            .copied()
294            .zip(corrected.iter().copied())
295            .collect()
296    }

331    #[test]
332    fn chain_collapses_within_its_component_only() {
333        // A chain A -> B -> H, split so the component boundary falls in the
334        // second batch. CCAA is one symbol from CAAA, and CAAA is one symbol from
335        // both CCAA and AAAA. Rows are sorted before the search, so CAAA meets
336        // AAAA, the chain's end, first. The other component is one edge between
337        // two distant UMIs, which has no collapse target. M and N are umi1 and
338        // every other UMI is umi2, so none sits on both sides.
339        let umi_m = umi("GGGG");
340        let umi_h = umi("AAAA");
341        let umi_a = umi("CCAA");
342        let umi_b = umi("CAAA");
343        let umi_n = umi("TTTT");
344        let batches = [
345            batch(&[umi_m, umi_m], &[umi_h, umi_a], &["cell", "cell"]),
346            batch(
347                &[umi_m, umi_n, umi("ACAC")],
348                &[umi_b, umi_h, umi("GTGT")],
349                &["cell", "cell", "other"],
350            ),
351        ];
352
353        let (table, stats) = run_graph_collapse(batches.into_iter().map(Ok), 2, 1, 1).unwrap();
354
355        assert_eq!(
356            collapsed_pairs(&table),
357            HashSet::from_iter([(umi_a, umi_h), (umi_b, umi_h)])
358        );
359        // "cell" has three leaves (A, B and N) and "other" has two. Only the
360        // chain in "cell" finds matches.
361        assert_eq!(stats.total_components, 2);
362        assert_eq!(stats.nodes_considered, 5);
363        assert_eq!(stats.targets_found, 2);
364        assert_eq!(stats.bfs_distance_counts(), [(2, 2)]);
365        assert_eq!(stats.hamming_distance_counts(), [(1, 2)]);
366    }
