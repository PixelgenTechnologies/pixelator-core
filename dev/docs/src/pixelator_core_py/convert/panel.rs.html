1use pyo3::exceptions::PyValueError;
2use pyo3::prelude::*;
3
4use pixelator_core::common::{Panel, PanelMarker};
5
6use super::common::reject_path_like;

8/// One panel row, extracted from a dict (`DataFrame.to_dicts()`).
9/// Extra columns (`control`, `uniprot_id`, …) are ignored.
10#[derive(FromPyObject)]
11#[pyo3(from_item_all)]
12struct PyMarker {
13    marker_id: String,
14    sequence_1: String,
15    sequence_2: String,
16}
17
18/// Build a [`Panel`] from a Python `PNAAntibodyPanel` (or any duck-typed equivalent).
19pub fn panel_from_py(ob: &Bound<'_, PyAny>) -> PyResult<Panel> {
20    reject_path_like(
21        ob,
22        "panel",
23        "PNAAntibodyPanel",
24        "Load it with `load_antibody_panel(...)` or `PNAAntibodyPanel.from_csv(...)`.",
25    )?;
26    let rows: Vec<PyMarker> = ob
27        .call_method0("to_polars")?
28        .call_method0("to_dicts")?
29        .extract()?;
30    Panel::from_markers(rows.into_iter().map(PanelMarker::from).collect())
31        .map_err(|e| PyValueError::new_err(e.to_string()))
32}
33
34impl From<PyMarker> for PanelMarker {
35    fn from(marker: PyMarker) -> Self {
36        Self {
37            marker_id: marker.marker_id,
38            sequence_1: marker.sequence_1,
39            sequence_2: marker.sequence_2,
40        }
41    }
42}
