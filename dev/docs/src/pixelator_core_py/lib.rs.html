11// MiMalloc backs every Rust allocation in this extension, including the
12// statically linked `pixelator-core` code, so the hot edgelist path keeps its
13// allocator performance even when driven from Python -- except on macOS, where
14// mimalloc reserves a fixed pthread thread-local-storage slot for the per-thread
15// heap pointer. PyArrow bundles its own, independently-built mimalloc that
16// reserves the same kind of slot, and when both are loaded into the same process
17// they clobber each other's per-thread heap pointer. Falling back to the system
18// allocator on macOS avoids that.
19#[cfg(not(target_os = "macos"))]
20#[global_allocator]
21static GLOBAL: mimalloc::MiMalloc = mimalloc::MiMalloc;
22
